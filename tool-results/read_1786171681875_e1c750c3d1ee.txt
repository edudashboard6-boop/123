   101→// -----------------------------------------------------------------------------
   102→// Component
   103→// -----------------------------------------------------------------------------
   104→
   105→export function SettingsPanel() {
   106→  const [settings, setSettings] = useState<AppSettings>(DEFAULT_SETTINGS);
   107→  const [loading, setLoading] = useState(true);
   108→  const [saving, setSaving] = useState(false);
   109→  const [saveFlash, setSaveFlash] = useState<string | null>(null);
   110→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   111→
   112→  // ---------------------------------------------------------------------------
   113→  // Fetch settings
   114→  // ---------------------------------------------------------------------------
   115→
   116→  const fetchSettings = useCallback(async () => {
   117→    setLoading(true);
   118→    try {
   119→      const res = await fetch('/api/settings');
   120→      if (res.ok) {
   121→        const data = await res.json();
   122→        setSettings({ ...DEFAULT_SETTINGS, ...data });
   123→      }
   124→    } catch {
   125→      // use defaults
   126→    } finally {
   127→      setLoading(false);
   128→    }
   129→  }, []);
   130→
   131→  useEffect(() => {
   132→    fetchSettings();
   133→  }, [fetchSettings]);
   134→
   135→  // ---------------------------------------------------------------------------
   136→  // Save settings (debounced)
   137→  // ---------------------------------------------------------------------------
   138→
   139→  const saveSettings = useCallback(async (newSettings: AppSettings) => {
   140→    setSaving(true);
   141→    try {
   142→      const res = await fetch('/api/settings', {
   143→        method: 'PATCH',
   144→        headers: { 'Content-Type': 'application/json' },
   145→        body: JSON.stringify(newSettings),
   146→      });
   147→      if (!res.ok) throw new Error('Save failed');
   148→    } catch {
   149→      toast.error('Failed to save settings');
   150→    } finally {
   151→      setSaving(false);
   152→    }
   153→  }, []);
   154→
   155→  const updateSetting = useCallback(
   156→    <K extends keyof AppSettings>(key: K, value: AppSettings[K]) => {
   157→      setSettings((prev) => {
   158→        const updated = { ...prev, [key]: value };
   159→        // Debounced save
   160→        if (debounceRef.current) clearTimeout(debounceRef.current);
   161→        debounceRef.current = setTimeout(() => {
   162→          saveSettings(updated);
   163→        }, 500);
   164→        return updated;
   165→      });
   166→    },
   167→    [saveSettings],
   168→  );
   169→
   170→  const updateAgentBudget = useCallback(
   171→    (agentId: string, value: number) => {
   172→      setSettings((prev) => {
   173→        const updated = {
   174→          ...prev,
   175→          agentBudgetLimits: { ...prev.agentBudgetLimits, [agentId]: value },
   176→        };
   177→        if (debounceRef.current) clearTimeout(debounceRef.current);
   178→        debounceRef.current = setTimeout(() => {
   179→          saveSettings(updated);
   180→        }, 500);
   181→        return updated;
   182→      });
   183→    },
   184→    [saveSettings],
   185→  );
   186→
   187→  const flashSave = useCallback((key: string) => {
   188→    setSaveFlash(key);
   189→    setTimeout(() => setSaveFlash(null), 1500);
   190→  }, []);
   191→
   192→  const handleResetCosts = useCallback(() => {
   193→    toast.success('Cost counters have been reset');
   194→  }, []);
   195→
   196→  // ---------------------------------------------------------------------------
   197→  // Render helpers
   198→  // ---------------------------------------------------------------------------
   199→
   200→  const renderSaveIndicator = (key: string) => {
   201→    if (saveFlash === key) {
   202→      return (
   203→        <span className="flex items-center gap-1 text-[10px] text-emerald-400">
   204→          <Check className="h-3 w-3" /> Saved
   205→        </span>
   206→      );
   207→    }
   208→    if (saving) {
   209→      return (
   210→        <span className="text-[10px] text-muted-foreground/70">Saving...</span>
   211→      );
   212→    }
   213→    return null;
   214→  };
   215→
   216→  // ---------------------------------------------------------------------------
   217→  // Render
   218→  // ---------------------------------------------------------------------------
   219→
   220→  return (
   221→    <TooltipProvider>
   222→      <div className="flex h-full flex-col min-h-0">
   223→        {/* Header */}

   225→          <div className="flex items-center justify-between">
   226→            <div className="flex items-center gap-3">
   227→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-violet-600/15 border border-violet-500/20">
   228→                <Settings className="h-4.5 w-4.5 text-violet-400" />
   229→              </div>
   230→              <div>
   231→                <h1 className="text-base font-semibold text-foreground">Settings</h1>
   232→                <p className="text-xs text-muted-foreground/70">Configure system preferences and orchestration parameters</p>
   233→              </div>
   234→            </div>
   235→          </div>
   236→        </div>
   237→
   238→        <ScrollArea className="flex-1 min-h-0">
   239→          <div className="mx-auto w-full max-w-3xl space-y-6 p-6">
   240→
   241→            {/* General Settings */}
   242→            <Card className="bg-card/80 border-border/60">
   243→              <CardHeader className="pb-3">
   244→                <CardTitle className="text-sm font-semibold text-secondary-foreground">General</CardTitle>
   245→                <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   246→                  Appearance, language, and default behavior
   247→                </CardDescription>
   248→              </CardHeader>
   249→              <CardContent className="space-y-5">
   250→                {/* Theme */}
   251→                <div className="flex items-center justify-between">
   252→                  <div className="flex items-center gap-3">
   253→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-muted/60">
   254→                      {settings.theme === 'dark' ? (
   255→                        <Moon className="h-4 w-4 text-violet-400" />
   256→                      ) : (
   257→                        <Sun className="h-4 w-4 text-amber-400" />
   258→                      )}
   259→                    </div>
   260→                    <div>
   261→                      <Label className="text-xs text-secondary-foreground">Theme</Label>
   262→                      <p className="text-[11px] text-muted-foreground/50">Switch between light and dark mode</p>
   263→                    </div>
   264→                  </div>
   265→                  <Select
   266→                    value={settings.theme}
   267→                    onValueChange={(v) => updateSetting('theme', v as 'dark' | 'light')}
   268→                  >
   269→                    <SelectTrigger className="w-32 h-8 bg-muted/60 border-input/60 text-secondary-foreground text-xs">
   270→                      <SelectValue />
   271→                    </SelectTrigger>
   272→                    <SelectContent className="bg-card border-border">
   273→                      <SelectItem value="dark" className="text-secondary-foreground">
   274→                        <div className="flex items-center gap-2"><Moon className="h-3 w-3" /> Dark</div>
   275→                      </SelectItem>

   311→                  </div>

   316→                    <SelectTrigger className="w-32 h-8 bg-muted/60 border-input/60 text-secondary-foreground text-xs">
   317→                      <SelectValue />
   318→                    </SelectTrigger>
   319→                    <SelectContent className="bg-card border-border">
   320→                      <SelectItem value="fast" className="text-secondary-foreground">Fast</SelectItem>

   322→                      <SelectItem value="redteam" className="text-secondary-foreground">Red Team</SelectItem>
   323→                    </SelectContent>
   324→                  </Select>
   325→                </div>
   326→
   327→                <Separator className="bg-muted/60" />
   328→
   329→                {/* Auto-save */}
   330→                <div className="flex items-center justify-between">
   331→                  <div>
   332→                    <Label className="text-xs text-secondary-foreground">Auto-save</Label>
   333→                    <p className="text-[11px] text-muted-foreground/50">Automatically save chat sessions</p>
   334→                  </div>
   335→                  <Switch
   336→                    checked={settings.autoSave}
   337→                    onCheckedChange={(v) => { updateSetting('autoSave', v); flashSave('autoSave'); }}
   338→                    className="data-[state=checked]:bg-violet-600"
   339→                  />
   340→                </div>
   341→              </CardContent>
   342→            </Card>
   343→
   344→            {/* Cost Control */}
   345→            <Card className="bg-card/80 border-border/60">
   346→              <CardHeader className="pb-3">
   347→                <div className="flex items-center justify-between">
   348→                  <div>
   349→                    <CardTitle className="text-sm font-semibold text-secondary-foreground">Cost Control</CardTitle>
   350→                    <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   351→                      Budget limits and spending alerts
   352→                    </CardDescription>
   353→                  </div>
   354→                  <Button
   355→                    variant="outline"
   356→                    size="sm"
   357→                    className="h-7 text-[11px] text-muted-foreground hover:text-secondary-foreground border-input hover:bg-muted gap-1.5"
   358→                    onClick={handleResetCosts}
   359→                  >
   360→                    <RotateCcw className="h-3 w-3" />
   361→                    Reset Counters

   363→                </div>
   364→              </CardHeader>
   365→              <CardContent className="space-y-5">
   366→                {/* Per-session budget */}
   367→                <div className="space-y-2">
   368→                  <div className="flex items-center justify-between">
   369→                    <Label className="text-xs text-secondary-foreground">Per-Session Budget Limit</Label>

   373→                    <span className="text-sm text-muted-foreground/70 font-medium">$</span>
   374→                    <Input
   375→                      type="number"
   376→                      min={0}
   377→                      step={1}
   378→                      value={settings.sessionBudgetLimit}
   379→                      onChange={(e) => updateSetting('sessionBudgetLimit', parseFloat(e.target.value) || 0)}
   380→                      onBlur={() => flashSave('sessionBudget')}
   381→                      className="h-8 w-24 bg-muted/60 border-input/60 text-secondary-foreground text-sm"
   382→                    />
   383→                    <span className="text-[11px] text-muted-foreground/50">per session</span>
   384→                  </div>
   385→                </div>
   386→
   387→                <Separator className="bg-muted/60" />
   388→
   389→                {/* Per-agent budget limits */}
   390→                <div className="space-y-3">
   391→                  <Label className="text-xs text-secondary-foreground">Per-Agent Budget Limits</Label>
   392→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   393→                    {Object.entries(AGENT_LABELS).map(([agentId, { name, color }]) => (

   395→                        <span className={`text-xs font-medium ${color}`}>{name}</span>

   400→                            min={0}
   401→                            step={0.5}
   402→                            value={settings.agentBudgetLimits[agentId] ?? 0}
   403→                            onChange={(e) => updateAgentBudget(agentId, parseFloat(e.target.value) || 0)}
   404→                            onBlur={() => flashSave(`agent-${agentId}`)}
   405→                            className="h-7 w-16 bg-muted/60 border-input/60 text-secondary-foreground text-xs text-right"
   406→                          />
   407→                        </div>

   422→                    value={[settings.budgetAlertThreshold]}
   423→                    min={50}

   425→                    step={5}
   426→                    onValueChange={([v]) => { updateSetting('budgetAlertThreshold', v); flashSave('threshold'); }}
   427→                    className="py-1"

   443→                  <div>
   444→                    <CardTitle className="text-sm font-semibold text-secondary-foreground">Security</CardTitle>
   445→                    <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   446→                      Sandbox, tool approvals, and audit settings
   447→                    </CardDescription>
   448→                  </div>
   449→                </div>
   450→              </CardHeader>
   451→              <CardContent className="space-y-5">

   459→                    checked={settings.sandboxMode}
   460→                    onCheckedChange={(v) => { updateSetting('sandboxMode', v); flashSave('sandbox'); }}
   461→                    className="data-[state=checked]:bg-violet-600"
   462→                  />
   463→                </div>
