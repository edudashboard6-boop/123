   288→// ---------- Tool Call Card ----------
   289→function ToolCallCard({ toolCall }: { toolCall: ToolCall }) {
   290→  const [copied, setCopied] = useState(false);
   291→  const statusIcon =
   292→    toolCall.status === 'success' ? (
   293→      <CheckCircle className="h-3.5 w-3.5 text-green-400" />
   294→    ) : toolCall.status === 'error' ? (
   295→      <XCircle className="h-3.5 w-3.5 text-red-400" />
   296→    ) : (
   297→      <Clock className="h-3.5 w-3.5 text-muted-foreground/70" />
   298→    );
   299→
   300→  const handleCopy = () => {
   301→    navigator.clipboard.writeText(toolCall.result ?? JSON.stringify(toolCall.arguments, null, 2));
   302→    setCopied(true);
   303→    setTimeout(() => setCopied(false), 2000);
   304→  };
   305→
   306→  return (
   307→    <Collapsible className="my-2">
   308→      <CollapsibleTrigger className="flex w-full items-center gap-2 rounded-lg border border-border bg-card/60 px-3 py-2 text-xs transition-colors hover:bg-muted/60 [&[data-state=open]>svg.chevron]:rotate-90">
   309→        <ChevronRight className="h-3 w-3 text-muted-foreground/70 chevron transition-transform" />
   310→        <Wrench className="h-3 w-3 text-violet-400" />
   311→        <span className="font-medium text-secondary-foreground">{toolCall.name}</span>
   312→        {statusIcon}
   313→        {toolCall.duration != null && (
   314→          <span className="ml-auto text-muted-foreground/50">{toolCall.duration}ms</span>
   315→        )}
   316→      </CollapsibleTrigger>
   317→      <CollapsibleContent className="mt-1 space-y-2 pl-6">
   318→        <div className="rounded-lg border border-border bg-card/40 p-3">
   319→          <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50 mb-1">Arguments</p>
   320→          <pre className="text-xs text-muted-foreground overflow-x-auto whitespace-pre-wrap break-all">
   321→            {JSON.stringify(toolCall.arguments, null, 2)}
   322→          </pre>
   323→        </div>
   324→        {toolCall.result && (
   325→          <div className="rounded-lg border border-border bg-card/40 p-3">
   326→            <div className="flex items-center justify-between mb-1">
   327→              <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50">Result</p>
   328→              <button
   329→                onClick={handleCopy}
   330→                className="text-muted-foreground/50 hover:text-muted-foreground transition-colors"
   331→              >
   332→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   333→              </button>
   334→            </div>
   335→            <pre className="text-xs text-muted-foreground overflow-x-auto whitespace-pre-wrap break-all max-h-48">
   336→              {toolCall.result}
   337→            </pre>
   338→          </div>
   339→        )}
   340→      </CollapsibleContent>
   341→    </Collapsible>
   342→  );
   343→}

   384→// ---------- Markdown Renderer ----------
   385→function MarkdownRenderer({ content }: { content: string }) {
   386→  return (
   387→    <ReactMarkdown
   388→      components={{
   389→        code({ className, children, ...props }) {
   390→          const match = /language-(\w+)/.exec(className || '');
   391→          const isInline = !match;
   392→          return isInline ? (
   393→            <code
   394→              className="rounded bg-muted px-1.5 py-0.5 text-xs text-violet-300 font-mono"
   395→              {...props}
   396→            >
   397→              {children}
   398→            </code>
   399→          ) : (
   400→            <div className="relative my-2 rounded-lg border border-border overflow-hidden">
   401→              <div className="flex items-center justify-between border-b border-border bg-card/80 px-4 py-1.5">
   402→                <span className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   403→                  {match[1]}
   404→                </span>
   405→              </div>
   406→              <SyntaxHighlighter
   407→                style={oneDark}
   408→                language={match[1]}
   409→                PreTag="div"
   410→                customStyle={{
   411→                  margin: 0,
   412→                  padding: '1rem',

   426→        ul({ children }) {

   434→        },

   446→            <blockquote className="border-l-2 border-violet-500/40 pl-3 my-2 text-muted-foreground italic">

   474→        strong({ children }) {
   475→          return <strong className="font-semibold text-foreground">{children}</strong>;

   484→// ---------- Message Bubble ----------
   485→function MessageBubble({ message }: { message: ChatMessage }) {
   486→  const isUser = message.role === 'user';
   487→  const isSystem = message.role === 'system';
   488→  const isAgent = !isUser && !isSystem;
   489→  const agentId = message.role as AgentId;
   490→  const AgentIcon = isAgent ? AGENT_ICON_MAP[agentId] : null;
   491→  const hasToolCalls = message.toolCalls && message.toolCalls.length > 0;
   492→  const hasDebateData = !!message.debateData;
   493→
   494→  // System message
   495→  if (isSystem) {
   496→    return (
   497→      <motion.div
   498→        initial={{ opacity: 0, y: 10 }}
   499→        animate={{ opacity: 1, y: 0 }}
   500→        className="flex justify-center py-2"
   501→      >
   502→        <span className="text-xs text-muted-foreground/70 bg-card/60 rounded-full px-3 py-1">
   503→          {message.content}
   504→        </span>
   505→      </motion.div>
   506→    );
   507→  }
   508→
   509→  // User message
   510→  if (isUser) {
   511→    return (
   512→      <motion.div
   513→        initial={{ opacity: 0, y: 10 }}
   514→        animate={{ opacity: 1, y: 0 }}
   515→        className="flex justify-end px-4 py-1.5"
   516→      >
   517→        <div className="max-w-[75%] rounded-2xl rounded-tr-md bg-muted px-4 py-3">

   519→            {message.content}
   520→          </p>
   521→        </div>
   522→      </motion.div>
   523→    );
   524→  }
   525→
   526→  // Agent message
   527→  return (
   528→    <motion.div
   529→      initial={{ opacity: 0, y: 10 }}
   530→      animate={{ opacity: 1, y: 0 }}
   531→      className="flex gap-3 px-4 py-1.5"
   532→    >
   533→      <div className="flex shrink-0 flex-col items-center gap-1">
   534→        <Avatar className="h-8 w-8 border border-border">
   535→          <AvatarFallback className={`${AGENT_AVATAR_BG[agentId]} text-white text-xs font-bold`}>
   536→            <AgentIcon className="h-4 w-4" />
   537→          </AvatarFallback>
   538→        </Avatar>
   539→      </div>
   540→
   541→      <div className={`flex-1 min-w-0 border-l-2 ${AGENT_BORDER_COLOR_MAP[agentId]} pl-3`}>

   597→// ---------- Debate Timeline ----------
   598→function DebateTimeline({
   599→  currentPhase,
   600→  messages,
   601→}: {
   602→  currentPhase?: DebatePhase;
   603→  messages: ChatMessage[];
   604→}) {
   605→  const phaseOrder: DebatePhase[] = ['draft', 'critique', 'rebuttal', 'arbitrate', 'verify', 'commit'];
   606→
   607→  // Determine which phases have been reached based on messages
   608→  const reachedPhases = new Set<DebatePhase>();
   609→  messages.forEach((m) => {
   610→    if (m.debateData) {
   611→      reachedPhases.add(m.debateData.phase);
   612→    }
   613→  });
   614→
   615→  const currentIndex = currentPhase ? phaseOrder.indexOf(currentPhase) : -1;
   616→
   617→  return (
   618→    <div className="space-y-1">
   619→      <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50 mb-2">Debate Timeline</p>
   620→      <div className="space-y-0">
   621→        {phaseOrder.map((phase, idx) => {
   622→          const phaseDef = DEBATE_PHASES.find((p) => p.phase === phase);
   623→          const isCompleted = reachedPhases.has(phase) && idx < currentIndex;
   624→          const isActive = phase === currentPhase;
   625→          const isPending = !reachedPhases.has(phase) && idx > currentIndex;
   626→          const Icon = PHASE_ICONS[phase];
   627→
   628→          return (
   629→            <div key={phase} className="flex items-start gap-2.5">
   630→              {/* Vertical line + node */}

   644→                    <Icon className="h-3.5 w-3.5" />

   676→// ---------- Cost Breakdown ----------
   677→function CostBreakdown({ messages }: { messages: ChatMessage[] }) {
   678→  const agentMap = new Map<string, { tokens: number; cost: number; calls: number }>();
   679→
   680→  messages.forEach((m) => {
   681→    if (m.role === 'system' || m.role === 'user') return;
   682→    const existing = agentMap.get(m.agentName) ?? { tokens: 0, cost: 0, calls: 0 };
   683→    agentMap.set(m.agentName, {
   684→      tokens: existing.tokens + m.tokens,
   685→      cost: existing.cost + m.cost,
   686→      calls: existing.calls + 1,
   687→    });
   688→  });
   689→
   690→  const entries = Array.from(agentMap.entries());
   691→  if (entries.length === 0) return null;
   692→
   693→  const maxTokens = Math.max(...entries.map(([, v]) => v.tokens), 1);
   694→
   695→  return (
   696→    <div className="space-y-2">
   697→      <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50">Cost Breakdown</p>
   698→      <div className="space-y-2.5">
   699→        {entries.map(([name, data]) => {
   700→          const agentDef = DEFAULT_AGENTS.find((a) => a.name === name);
   701→          const pct = (data.tokens / maxTokens) * 100;
   702→          return (
   703→            <div key={name} className="space-y-1">
   704→              <div className="flex items-center justify-between">
   705→                <div className="flex items-center gap-2">
   706→                  <div
   707→                    className={`h-2 w-2 rounded-full ${
   708→                      agentDef ? agentDef.color.replace('text-', 'bg-') : 'bg-accent'
   709→                    }`}
   710→                  />
   711→                  <span className="text-xs text-secondary-foreground">{name}</span>
   712→                </div>
   713→                <div className="flex items-center gap-2 text-[10px] text-muted-foreground/70">
   714→                  <span>{data.tokens.toLocaleString()} tok</span>
   715→                  <span>${data.cost.toFixed(4)}</span>
   716→                </div>
   717→              </div>
   718→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   719→                <div
   720→                  className={`h-full rounded-full transition-all ${
   721→                    agentDef ? agentDef.color.replace('text-', 'bg-') : 'bg-accent'
   722→                  }`}
   723→                  style={{ width: `${pct}%` }}
   724→                />
   725→              </div>
   726→            </div>
   727→          );
   728→        })}
   729→      </div>
   730→    </div>
   731→  );
   732→}
