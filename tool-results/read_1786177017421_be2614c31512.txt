   131→   131→
   132→   132→export function ToolsPanel() {
   133→   133→  const [servers, setServers] = useState<McpServer[]>([]);
   134→   134→  const [loading, setLoading] = useState(true);
   135→   135→  const [toolToggles, setToolToggles] = useState<Record<string, boolean>>(() => {
   136→   136→    const init: Record<string, boolean> = {};
   137→   137→    BUILT_IN_TOOLS.forEach((t) => { init[t.id] = true; });
   138→   138→    return init;
   139→   139→  });
   140→   140→
   141→   141→  // Add/Edit dialog
   142→   142→  const [dialogOpen, setDialogOpen] = useState(false);
   143→   143→  const [editingServer, setEditingServer] = useState<McpServer | null>(null);
   144→   144→  const [saving, setSaving] = useState(false);
   145→   145→  const [form, setForm] = useState({
   146→   146→    name: '',
   147→   147→    description: '',
   148→   148→    transport: 'stdio' as TransportType,
   149→   149→    command: '',
   150→   150→    args: '',
   151→   151→    url: '',
   152→   152→    envVars: [{ key: '', value: '' }],
   153→   153→    isActive: true,
   154→   154→  });
   155→   155→
   156→   156→  // Delete confirmation
   157→   157→  const [deleteTarget, setDeleteTarget] = useState<McpServer | null>(null);
   158→   158→  const [deleting, setDeleting] = useState(false);
   159→   159→
   160→   160→  // ---------------------------------------------------------------------------
   161→   161→  // Fetch MCP servers
   162→   162→  // ---------------------------------------------------------------------------
   163→   163→
   164→   164→  const fetchServers = useCallback(async () => {
   165→   165→    setLoading(true);
   166→   166→    try {
   167→   167→      const res = await fetch('/api/mcp-servers');
   168→   168→      if (!res.ok) throw new Error('Failed to fetch servers');
   169→   169→      const data: McpServer[] = await res.json();
   170→   170→      setServers(data);
   171→   171→    } catch {
   172→   172→      setServers([]);
   173→   173→    } finally {
   174→   174→      setLoading(false);
   175→   175→    }
   176→   176→  }, []);
   177→   177→
   178→   178→  useEffect(() => {
   179→   179→    fetchServers();
   180→   180→  }, [fetchServers]);
   181→   181→
   182→   182→  // ---------------------------------------------------------------------------
   183→   183→  // Group built-in tools by category
   184→   184→  // ---------------------------------------------------------------------------
   185→   185→
   186→   186→  const toolsByCategory = useMemo(() => {
   187→   187→    const map = new Map<string, typeof BUILT_IN_TOOLS>();
   188→   188→    BUILT_IN_TOOLS.forEach((tool) => {
   189→   189→      const cat = tool.category;
   190→   190→      if (!map.has(cat)) map.set(cat, []);
   191→   191→      map.get(cat)!.push(tool);
   192→   192→    });
   193→   193→    return map;
   194→   194→  }, []);
   195→   195→
   196→   196→  // ---------------------------------------------------------------------------
   197→   197→  // Dialog helpers
   198→   198→  // ---------------------------------------------------------------------------
   199→   199→
   200→   200→  const openAddDialog = useCallback(() => {
   201→   201→    setEditingServer(null);
   202→   202→    setForm({
   203→   203→      name: '',
   204→   204→      description: '',
   205→   205→      transport: 'stdio',
   206→   206→      command: '',
   207→   207→      args: '',
   208→   208→      url: '',
   209→   209→      envVars: [{ key: '', value: '' }],
   210→   210→      isActive: true,
   211→   211→    });
   212→   212→    setDialogOpen(true);
   213→   213→  }, []);
   214→   214→
   215→   215→  const openEditDialog = useCallback((server: McpServer) => {
   216→   216→    setEditingServer(server);
   217→   217→    setForm({
   218→   218→      name: server.name,
   219→   219→      description: server.description,
   220→   220→      transport: server.transport,
   221→   221→      command: server.command,
   222→   222→      args: server.args.join(', '),
   223→   223→      url: server.url,
   224→   224→      envVars: Object.entries(server.envVars).map(([key, value]) => ({ key, value })),
   225→   225→      isActive: server.isActive,
   226→   226→    });
   227→   227→    setDialogOpen(true);
   228→   228→  }, []);
   229→   229→
   230→   230→  const updateForm = useCallback(<K extends keyof typeof form>(key: K, value: (typeof form)[K]) => {
   231→   231→    setForm((prev) => ({ ...prev, [key]: value }));
   232→   232→  }, []);
   233→   233→
   234→   234→  // ---------------------------------------------------------------------------
   235→   235→  // Save (create or update)
   236→   236→  // ---------------------------------------------------------------------------
   237→   237→
   238→   238→  const handleSave = useCallback(async () => {
   239→   239→    if (!form.name.trim()) {
   240→   240→      toast.error('Server name is required');
   241→   241→      return;
   242→   242→    }
   243→   243→
   244→   244→    const envObj: Record<string, string> = {};
   245→   245→    form.envVars.forEach(({ key, value }) => {
   246→   246→      if (key.trim()) envObj[key.trim()] = value;
   247→   247→    });
   248→   248→
   249→   249→    const payload = {
   250→   250→      name: form.name.trim(),
   251→   251→      description: form.description.trim(),
   252→   252→      transport: form.transport,
   253→   253→      command: form.command.trim(),
   254→   254→      args: form.args.split(',').map((a) => a.trim()).filter(Boolean),
   255→   255→      url: form.url.trim(),
   256→   256→      envVars: envObj,
   257→   257→      isActive: form.isActive,
   258→   258→    };
   259→   259→
   260→   260→    setSaving(true);
   261→   261→    try {
   262→   262→      const isEdit = !!editingServer;
   263→   263→      const url = isEdit
   264→   264→        ? `/api/mcp-servers/${editingServer.id}`
   265→   265→        : '/api/mcp-servers';
   266→   266→      const method = isEdit ? 'PATCH' : 'POST';
   267→   267→
   268→   268→      const res = await fetch(url, {
   269→   269→        method,
   270→   270→        headers: { 'Content-Type': 'application/json' },
   271→   271→        body: JSON.stringify(payload),
   272→   272→      });
   273→   273→
   274→   274→      if (!res.ok) {
   275→   275→        const body = await res.json().catch(() => ({}));
   276→   276→        throw new Error(body.error ?? `Failed to ${isEdit ? 'update' : 'create'} server`);
   277→   277→      }
   278→   278→
   279→   279→      const saved: McpServer = await res.json();
   280→   280→      if (isEdit) {
   281→   281→        setServers((prev) => prev.map((s) => (s.id === saved.id ? saved : s)));
   282→   282→        toast.success(`Updated ${saved.name}`);
   283→   283→      } else {
   284→   284→        setServers((prev) => [...prev, saved]);
   285→   285→        toast.success(`Added ${saved.name}`);
   286→   286→      }
   287→   287→      setDialogOpen(false);
   288→   288→    } catch (err) {
   289→   289→      toast.error(err instanceof Error ? err.message : 'Save failed');
   290→   290→    } finally {
   291→   291→      setSaving(false);
   292→   292→    }
   293→   293→  }, [form, editingServer]);
   294→   294→
   295→   295→  // ---------------------------------------------------------------------------
   296→   296→  // Delete
   297→   297→  // ---------------------------------------------------------------------------
   298→   298→
   299→   299→  const handleDelete = useCallback(async () => {
   300→   300→    if (!deleteTarget) return;
   301→   301→    setDeleting(true);
   302→   302→    try {
   303→   303→      const res = await fetch(`/api/mcp-servers/${deleteTarget.id}`, { method: 'DELETE' });
   304→   304→      if (!res.ok) throw new Error('Delete failed');
   305→   305→      setServers((prev) => prev.filter((s) => s.id !== deleteTarget.id));
   306→   306→      toast.success(`Deleted ${deleteTarget.name}`);
   307→   307→    } catch {
   308→   308→      toast.error('Failed to delete server');
   309→   309→    } finally {
   310→   310→      setDeleting(false);
   311→   311→      setDeleteTarget(null);
   312→   312→    }
   313→   313→  }, [deleteTarget]);
   314→   314→
   315→   315→  // ---------------------------------------------------------------------------
   316→   316→  // Render
   317→   317→  // ---------------------------------------------------------------------------
   318→   318→
   319→   319→  return (
   320→   320→    <TooltipProvider>
   321→   321→      <div className="flex h-full flex-col min-h-0">
   322→   322→        {/* Header */}
   323→   323→        <div className="flex flex-col gap-1 border-b border-border/60 px-6 py-4">
   324→   324→          <div className="flex items-center justify-between">
   325→   325→            <div className="flex items-center gap-3">
   326→   326→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-violet-600/15 border border-violet-500/20">
   327→   327→                <Wrench className="h-4.5 w-4.5 text-violet-400" />
   328→   328→              </div>
   329→   329→              <div>
   330→   330→                <h1 className="text-base font-semibold text-foreground">Tools & Integrations</h1>
   331→   331→                <p className="text-xs text-muted-foreground/70">Configure MCP servers and tool permissions</p>
   332→   332→              </div>
   333→   333→            </div>
   334→   334→            <Button
   335→   335→              size="sm"
   336→   336→              className="bg-violet-600 hover:bg-violet-500 text-white gap-1.5"
   337→   337→              onClick={openAddDialog}
   338→   338→            >
   339→   339→              <Plus className="h-3.5 w-3.5" />
   340→   340→              Add Server
   341→   341→            </Button>
   342→   342→          </div>
   343→   343→        </div>
   344→   344→
   345→   345→        <ScrollArea className="flex-1 min-h-0">
   346→   346→          <div className="mx-auto w-full max-w-5xl space-y-6 p-6">
   347→   347→
   348→   348→            {/* Built-in Tools */}
   349→   349→            <Card className="bg-card/80 border-border/60">
   350→   350→              <CardHeader className="pb-3">
   351→   351→                <div>
   352→   352→                  <CardTitle className="text-sm font-semibold text-secondary-foreground">Built-in Tools</CardTitle>
   353→   353→                  <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   354→   354→                    {BUILT_IN_TOOLS.length} tools available to agents. Toggle to enable or disable.
   355→   355→                  </CardDescription>
   356→   356→                </div>
   357→   357→              </CardHeader>
   358→   358→              <CardContent className="pt-0">
   359→   359→                <Accordion type="multiple" defaultValue={Object.keys(toolsByCategory)} className="w-full">
   360→   360→                  {Array.from(toolsByCategory.entries()).map(([category, tools]) => (
   361→   361→                    <AccordionItem key={category} value={category} className="border-border/40">
   362→   362→                      <AccordionTrigger className="py-2.5 hover:no-underline">
   363→   363→                        <div className="flex items-center gap-2">
   364→   364→                          <Badge
   365→   365→                            variant="outline"
   366→   366→                            className={`text-[10px] h-4 px-1.5 border ${CATEGORY_COLORS[category] ?? CATEGORY_COLORS.system}`}
   367→   367→                          >
   368→   368→                            {CATEGORY_LABELS[category] ?? category}
   369→   369→                          </Badge>
   370→   370→                          <span className="text-xs text-muted-foreground/70">{tools.length} tools</span>
   371→   371→                        </div>
   372→   372→                      </AccordionTrigger>
   373→   373→                      <AccordionContent>
   374→   374→                        <div className="space-y-1">
   375→   375→                          {tools.map((tool) => {
   376→   376→                            const Icon = toolIconForId(tool.id);
   377→   377→                            return (
   378→   378→                              <div
   379→   379→                                key={tool.id}
   380→   380→                                className="group flex items-center justify-between gap-3 rounded-lg px-3 py-2 hover:"
   381→   381→                              >
   382→   382→                                <div className="flex items-center gap-3 min-w-0">
   383→   383→                                  <div className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md">
   384→   384→                                    <Icon className="h-3.5 w-3.5 text-muted-foreground" />
   385→   385→                                  </div>
   386→   386→                                  <div className="min-w-0">
   387→   387→                                    <p className="text-xs font-medium text-secondary-foreground">{tool.name}</p>
   388→   388→                                    <p className="text-[11px] text-muted-foreground/50 truncate max-w-md">{tool.description}</p>
   389→   389→                                  </div>
   390→   390→                                </div>
   391→   391→                                <Switch
   392→   392→                                  checked={toolToggles[tool.id] ?? true}
   393→   393→                                  onCheckedChange={(checked) =>
   394→   394→                                    setToolToggles((prev) => ({ ...prev, [tool.id]: checked }))
   395→   395→                                  }
   396→   396→                                  className="data-[state=checked]:bg-violet-600"
   397→   397→                                />
   398→   398→                              </div>
   399→   399→                            );
   400→   400→                          })}
   401→   401→                        </div>
   402→   402→                      </AccordionContent>
   403→   403→                    </AccordionItem>
   404→   404→                  ))}
   405→   405→                </Accordion>
   406→   406→              </CardContent>
   407→   407→            </Card>
   408→   408→
   409→   409→            {/* MCP Servers */}
   410→   410→            <div>
   411→   411→              <div className="flex items-center justify-between mb-3">
   412→   412→                <h2 className="text-sm font-semibold text-secondary-foreground">MCP Servers</h2>
   413→   413→                <span className="text-xs text-muted-foreground/70">{servers.length} configured</span>
   414→   414→              </div>
   415→   415→
   416→   416→              {loading ? (
   417→   417→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   418→   418→                  {Array.from({ length: 2 }).map((_, i) => (
   419→   419→                    <Skeleton key={i} className="h-40 rounded-xl bg-card/60" />
   420→   420→                  ))}
   421→   421→                </div>
   422→   422→              ) : servers.length === 0 ? (
   423→   423→                <div className="flex flex-col items-center justify-center py-16">
   424→   424→                  <div className="flex h-14 w-14 items-center justify-center rounded-2xl">
   425→   425→                    <Server className="h-6 w-6 text-muted-foreground/50" />
   426→   426→                  </div>
   427→   427→                  <p className="text-sm font-medium text-muted-foreground">No MCP servers configured</p>
   428→   428→                  <p className="mt-1 text-xs text-muted-foreground/50 max-w-xs text-center">
   429→   429→                    Add MCP servers to extend agent capabilities with external tools and services.
   430→   430→                  </p>
   431→   431→                </div>
   432→   432→              ) : (
   433→   433→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   434→   434→                  {servers.map((server) => (
   435→   435→                    <Card
   436→   436→                      key={server.id}
   437→   437→                      className="group relative bg-card/80 border-border/60 hover:border-violet-500/30 hover:shadow-[0_0_20px_rgba(139,92,246,0.06)] transition-all duration-200 overflow-hidden"
   438→   438→                    >
   439→   439→                      <CardContent className="p-4">
   440→   440→                        <div className="flex items-start justify-between gap-2 mb-3">
   441→   441→                          <div className="flex items-center gap-2.5 min-w-0">
   442→   442→                            <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg">
   443→   443→                              <Server className="h-4 w-4 text-muted-foreground" />
   444→   444→                            </div>
   445→   445→                            <div className="min-w-0">
   446→   446→                              <p className="text-sm font-medium text-secondary-foreground truncate" title={server.name}>
   447→   447→                                {server.name}
   448→   448→                              </p>
   449→   449→                              <div className="flex items-center gap-1.5 mt-0.5">
   450→   450→                                <span
   451→   451→                                  className={`inline-block h-1.5 w-1.5 rounded-full ${
   452→   452→                                    server.isActive ? 'bg-emerald-500' : 'bg-red-500'
   453→   453→                                  }`}
   454→   454→                                />
   455→   455→                                <span className="text-[11px] text-muted-foreground/70">
   456→   456→                                  {server.isActive ? 'Active' : 'Inactive'}
   457→   457→                                </span>
   458→   458→                              </div>
   459→   459→                            </div>
   460→   460→                          </div>
   461→   461→                          <div className="flex items-center gap-0.5 shrink-0 opacity-0 group-hover:opacity-100 transition-opacity">
   462→   462→                            <Tooltip>
   463→   463→                              <TooltipTrigger asChild>
   464→   464→                                <Button
   465→   465→                                  variant="ghost"
   466→   466→                                  size="icon"
   467→   467→                                  className="h-7 w-7 text-muted-foreground/70 hover:text-secondary-foreground hover:"
