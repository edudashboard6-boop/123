   100→// -----------------------------------------------------------------------------
   101→// Component
   102→// -----------------------------------------------------------------------------
   103→
   104→export function SettingsPanel() {
   105→  const [settings, setSettings] = useState<AppSettings>(DEFAULT_SETTINGS);
   106→  const [loading, setLoading] = useState(true);
   107→  const [saving, setSaving] = useState(false);
   108→  const [saveFlash, setSaveFlash] = useState<string | null>(null);
   109→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   110→
   111→  // ---------------------------------------------------------------------------
   112→  // Fetch settings
   113→  // ---------------------------------------------------------------------------
   114→
   115→  const fetchSettings = useCallback(async () => {
   116→    setLoading(true);
   117→    try {
   118→      const res = await fetch('/api/settings');
   119→      if (res.ok) {
   120→        const data = await res.json();
   121→        setSettings({ ...DEFAULT_SETTINGS, ...data });
   122→      }
   123→    } catch {
   124→      // use defaults
   125→    } finally {
   126→      setLoading(false);
   127→    }
   128→  }, []);
   129→
   130→  useEffect(() => {
   131→    fetchSettings();
   132→  }, [fetchSettings]);
   133→
   134→  // ---------------------------------------------------------------------------
   135→  // Save settings (debounced)
   136→  // ---------------------------------------------------------------------------
   137→
   138→  const saveSettings = useCallback(async (newSettings: AppSettings) => {
   139→    setSaving(true);
   140→    try {
   141→      const res = await fetch('/api/settings', {
   142→        method: 'PATCH',
   143→        headers: { 'Content-Type': 'application/json' },
   144→        body: JSON.stringify(newSettings),
   145→      });
   146→      if (!res.ok) throw new Error('Save failed');
   147→    } catch {
   148→      toast.error('Failed to save settings');
   149→    } finally {
   150→      setSaving(false);
   151→    }
   152→  }, []);
   153→
   154→  const updateSetting = useCallback(
   155→    <K extends keyof AppSettings>(key: K, value: AppSettings[K]) => {
   156→      setSettings((prev) => {
   157→        const updated = { ...prev, [key]: value };
   158→        // Debounced save
   159→        if (debounceRef.current) clearTimeout(debounceRef.current);
   160→        debounceRef.current = setTimeout(() => {
   161→          saveSettings(updated);
   162→        }, 500);
   163→        return updated;
   164→      });
   165→    },
   166→    [saveSettings],
   167→  );
   168→
   169→  const updateAgentBudget = useCallback(
   170→    (agentId: string, value: number) => {
   171→      setSettings((prev) => {
   172→        const updated = {
   173→          ...prev,
   174→          agentBudgetLimits: { ...prev.agentBudgetLimits, [agentId]: value },
   175→        };
   176→        if (debounceRef.current) clearTimeout(debounceRef.current);
   177→        debounceRef.current = setTimeout(() => {
   178→          saveSettings(updated);
   179→        }, 500);
   180→        return updated;
   181→      });
   182→    },
   183→    [saveSettings],
   184→  );
   185→
   186→  const flashSave = useCallback((key: string) => {
   187→    setSaveFlash(key);
   188→    setTimeout(() => setSaveFlash(null), 1500);
   189→  }, []);
   190→
   191→  const handleResetCosts = useCallback(() => {
   192→    toast.success('Cost counters have been reset');
   193→  }, []);
   194→
   195→  // ---------------------------------------------------------------------------
   196→  // Render helpers
   197→  // ---------------------------------------------------------------------------
   198→
   199→  const renderSaveIndicator = (key: string) => {
   200→    if (saveFlash === key) {
   201→      return (
   202→        <span className="flex items-center gap-1 text-[10px] text-emerald-400">
   203→          <Check className="h-3 w-3" /> Saved
   204→        </span>
   205→      );
   206→    }
   207→    if (saving) {
   208→      return (
   209→        <span className="text-[10px] text-zinc-500">Saving...</span>
   210→      );
   211→    }
   212→    return null;
   213→  };
   214→
   215→  // ---------------------------------------------------------------------------
   216→  // Render
   217→  // ---------------------------------------------------------------------------
   218→
   219→  return (
   220→    <TooltipProvider>
   221→      <div className="flex h-full flex-col min-h-0">
   222→        {/* Header */}

   224→          <div className="flex items-center justify-between">
   225→            <div className="flex items-center gap-3">
   226→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-violet-600/15 border border-violet-500/20">
   227→                <Settings className="h-4.5 w-4.5 text-violet-400" />
   228→              </div>
   229→              <div>
   230→                <h1 className="text-base font-semibold text-zinc-100">Settings</h1>
   231→                <p className="text-xs text-zinc-500">Configure system preferences and orchestration parameters</p>
   232→              </div>
   233→            </div>
   234→          </div>
   235→        </div>
   236→
   237→        <ScrollArea className="flex-1 min-h-0">
   238→          <div className="mx-auto w-full max-w-3xl space-y-6 p-6">
   239→
   240→            {/* General Settings */}
   241→            <Card className="bg-zinc-900/80 border-zinc-800/60">
   242→              <CardHeader className="pb-3">
   243→                <CardTitle className="text-sm font-semibold text-zinc-200">General</CardTitle>
   244→                <CardDescription className="text-xs text-zinc-500 mt-0.5">
   245→                  Appearance, language, and default behavior
   246→                </CardDescription>
   247→              </CardHeader>
   248→              <CardContent className="space-y-5">
   249→                {/* Theme */}
   250→                <div className="flex items-center justify-between">
   251→                  <div className="flex items-center gap-3">
   252→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-zinc-800/60">
   253→                      {settings.theme === 'dark' ? (
   254→                        <Moon className="h-4 w-4 text-violet-400" />
   255→                      ) : (
   256→                        <Sun className="h-4 w-4 text-amber-400" />
   257→                      )}
   258→                    </div>
   259→                    <div>
   260→                      <Label className="text-xs text-zinc-300">Theme</Label>
   261→                      <p className="text-[11px] text-zinc-600">Switch between light and dark mode</p>
   262→                    </div>
   263→                  </div>
   264→                  <Select
   265→                    value={settings.theme}
   266→                    onValueChange={(v) => updateSetting('theme', v as 'dark' | 'light')}
   267→                  >
   268→                    <SelectTrigger className="w-32 h-8 bg-zinc-800/60 border-zinc-700/60 text-zinc-300 text-xs">
   269→                      <SelectValue />
   270→                    </SelectTrigger>
   271→                    <SelectContent className="bg-zinc-900 border-zinc-800">
   272→                      <SelectItem value="dark" className="text-zinc-300">
   273→                        <div className="flex items-center gap-2"><Moon className="h-3 w-3" /> Dark</div>
   274→                      </SelectItem>

   314→                  >
   315→                    <SelectTrigger className="w-32 h-8 bg-zinc-800/60 border-zinc-700/60 text-zinc-300 text-xs">
   316→                      <SelectValue />
   317→                    </SelectTrigger>
   318→                    <SelectContent className="bg-zinc-900 border-zinc-800">
   319→                      <SelectItem value="fast" className="text-zinc-300">Fast</SelectItem>
   320→                      <SelectItem value="council" className="text-zinc-300">Council</SelectItem>
   321→                      <SelectItem value="redteam" className="text-zinc-300">Red Team</SelectItem>
   322→                    </SelectContent>
   323→                  </Select>
   324→                </div>
   325→
   326→                <Separator className="bg-zinc-800/60" />
   327→
   328→                {/* Auto-save */}
   329→                <div className="flex items-center justify-between">
   330→                  <div>
   331→                    <Label className="text-xs text-zinc-300">Auto-save</Label>
   332→                    <p className="text-[11px] text-zinc-600">Automatically save chat sessions</p>
   333→                  </div>
   334→                  <Switch
   335→                    checked={settings.autoSave}
   336→                    onCheckedChange={(v) => { updateSetting('autoSave', v); flashSave('autoSave'); }}
   337→                    className="data-[state=checked]:bg-violet-600"
   338→                  />
   339→                </div>
   340→              </CardContent>
   341→            </Card>
   342→
   343→            {/* Cost Control */}
   344→            <Card className="bg-zinc-900/80 border-zinc-800/60">
   345→              <CardHeader className="pb-3">
   346→                <div className="flex items-center justify-between">
   347→                  <div>
   348→                    <CardTitle className="text-sm font-semibold text-zinc-200">Cost Control</CardTitle>
   349→                    <CardDescription className="text-xs text-zinc-500 mt-0.5">
   350→                      Budget limits and spending alerts
   351→                    </CardDescription>
   352→                  </div>
   353→                  <Button
   354→                    variant="outline"
   355→                    size="sm"
   356→                    className="h-7 text-[11px] text-zinc-400 hover:text-zinc-200 border-zinc-700 hover:bg-zinc-800 gap-1.5"
   357→                    onClick={handleResetCosts}

   359→                    <RotateCcw className="h-3 w-3" />

   362→                </div>
   363→              </CardHeader>
   364→              <CardContent className="space-y-5">
   365→                {/* Per-session budget */}

   368→                    <Label className="text-xs text-zinc-300">Per-Session Budget Limit</Label>

   373→                    <Input
   374→                      type="number"
   375→                      min={0}
   376→                      step={1}
   377→                      value={settings.sessionBudgetLimit}
   378→                      onChange={(e) => updateSetting('sessionBudgetLimit', parseFloat(e.target.value) || 0)}
   379→                      onBlur={() => flashSave('sessionBudget')}
   380→                      className="h-8 w-24 bg-zinc-800/60 border-zinc-700/60 text-zinc-200 text-sm"
   381→                    />
   382→                    <span className="text-[11px] text-zinc-600">per session</span>
   383→                  </div>

   386→                <Separator className="bg-zinc-800/60" />
   387→

   389→                <div className="space-y-3">
   390→                  <Label className="text-xs text-zinc-300">Per-Agent Budget Limits</Label>

   392→                    {Object.entries(AGENT_LABELS).map(([agentId, { name, color }]) => (

   445→                      Sandbox, tool approvals, and audit settings

   456→                  </div>
   457→                  <Switch
   458→                    checked={settings.sandboxMode}

   460→                    className="data-[state=checked]:bg-violet-600"
