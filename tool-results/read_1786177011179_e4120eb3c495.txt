   459→// =============================================================================
   460→// Provider Form Dialog — Two-step wizard: (1) Connection → (2) Model selection
   461→// =============================================================================
   462→
   463→function ProviderFormDialog({
   464→  open,
   465→  onOpenChange,
   466→  editingProvider,
   467→  onSave,
   468→}: {
   469→  open: boolean;
   470→  onOpenChange: (open: boolean) => void;
   471→  editingProvider: LlmProvider | null;
   472→  onSave: (data: ProviderFormData) => Promise<void>;
   473→}) {
   474→  const [form, setForm] = useState<ProviderFormData>(EMPTY_FORM);
   475→  const [showApiKey, setShowApiKey] = useState(false);
   476→  const [saving, setSaving] = useState(false);
   477→  const [step, setStep] = useState<1 | 2>(1);
   478→
   479→  // Step 2 state — model fetching/selection
   480→  const [fetchedModels, setFetchedModels] = useState<string[]>([]);
   481→  const [selectedModels, setSelectedModels] = useState<string[]>([]);
   482→  const [fetching, setFetching] = useState(false);
   483→  const [fetchError, setFetchError] = useState('');
   484→  const [customModel, setCustomModel] = useState('');
   485→
   486→  const isEditing = editingProvider !== null;
   487→  const isLocal = PROVIDER_TYPES.find((t) => t.value === form.type)?.isLocal ?? false;
   488→
   489→  // Populate form when editing or resetting
   490→  useEffect(() => {
   491→    if (editingProvider) {
   492→      setForm({
   493→        name: editingProvider.name,
   494→        type: editingProvider.type,
   495→        baseUrl: editingProvider.baseUrl,
   496→        apiKey: editingProvider.apiKey,
   497→        models: editingProvider.models,
   498→        isActive: editingProvider.isActive,
   499→      });
   500→      setStep(1); // Editing stays on step 1 (models configured via separate dialog)
   501→    } else {
   502→      setForm(EMPTY_FORM);
   503→      setStep(1);
   504→    }
   505→    setShowApiKey(false);
   506→    setFetchedModels([]);
   507→    setSelectedModels([]);
   508→    setFetchError('');
   509→    setCustomModel('');
   510→  }, [editingProvider, open]);
   511→
   512→  // Auto-fill base URL when type changes
   513→  const handleTypeChange = useCallback(
   514→    (value: string) => {
   515→      const type = value as ProviderType;
   516→      const def = PROVIDER_TYPES.find((t) => t.value === type);
   517→      setForm((prev) => ({
   518→        ...prev,
   519→        type,
   520→        baseUrl: def?.defaultBaseUrl ?? '',
   521→      }));
   522→    },
   523→    []
   524→  );
   525→
   526→  // --- Step 2: Fetch models from the provider API ---
   527→  const handleFetchModels = useCallback(async () => {
   528→    setFetching(true);
   529→    setFetchError('');
   530→    try {
   531→      const res = await fetch('/api/providers/probe-models', {
   532→        method: 'POST',
   533→        headers: { 'Content-Type': 'application/json' },
   534→        body: JSON.stringify({ type: form.type, baseUrl: form.baseUrl, apiKey: form.apiKey }),
   535→      });
   536→      if (!res.ok) {
   537→        const body = await res.json().catch(() => ({}));
   538→        throw new Error(body.error ?? 'Failed to fetch models');
   539→      }
   540→      const data = await res.json();
   541→      const models: string[] = data.models;
   542→      if (models.length === 0) {
   543→        setFetchError('No models found. The API responded but returned an empty list.');
   544→      } else {
   545→        setFetchedModels(models);
   546→        setSelectedModels(models); // Auto-select all
   547→      }
   548→    } catch (err) {
   549→      setFetchError(err instanceof Error ? err.message : 'Failed to fetch models');
   550→    } finally {
   551→      setFetching(false);
   552→    }
   553→  }, [form.type, form.baseUrl, form.apiKey]);
   554→
   555→  const toggleModel = useCallback((model: string) => {
   556→    setSelectedModels((prev) =>
   557→      prev.includes(model) ? prev.filter((m) => m !== model) : [...prev, model]
   558→    );
   559→  }, []);
   560→
   561→  const addCustomModel = useCallback(() => {
   562→    const trimmed = customModel.trim();
   563→    if (trimmed && !selectedModels.includes(trimmed)) {
   564→      setSelectedModels((prev) => [...prev, trimmed]);
   565→      setCustomModel('');
   566→    }
   567→  }, [customModel, selectedModels]);
   568→
   569→  const allModels = useMemo(() => {
   570→    return [...new Set([...fetchedModels, ...selectedModels])].sort();
   571→  }, [fetchedModels, selectedModels]);
   572→
   573→  // --- Validate step 1 and advance to step 2 ---
   574→  const handleNext = useCallback(() => {
   575→    if (!form.name.trim()) {
   576→      toast.error('Provider name is required');

   938→// Model Configuration Dialog — Step 2: Fetch & select models for a provider

   941→function ModelConfigDialog({
   942→  open,
   943→  onOpenChange,
   944→  provider,
   945→  onUpdate,
   946→}: {
   947→  open: boolean;
   948→  onOpenChange: (open: boolean) => void;
   949→  provider: LlmProvider | null;
   950→  onUpdate: (id: string, data: { models: string[] }) => Promise<void>;
   951→}) {
   952→  const [selectedModels, setSelectedModels] = useState<string[]>([]);
   953→  const [fetchedModels, setFetchedModels] = useState<string[]>([]);
   954→  const [customModel, setCustomModel] = useState('');
   955→  const [fetching, setFetching] = useState(false);
   956→  const [saving, setSaving] = useState(false);
   957→  const [fetchError, setFetchError] = useState('');
   958→
   959→  // Sync selected models with provider when dialog opens
   960→  useEffect(() => {
   961→    if (provider) {
   962→      setSelectedModels(provider.models);
   963→      setFetchedModels([]);
   964→      setFetchError('');
   965→      setCustomModel('');
   966→    }
   967→  }, [provider, open]);
   968→
   969→  const allModels = useMemo(() => {
   970→    const combined = [...new Set([...fetchedModels, ...selectedModels])];
   971→    return combined.sort();
   972→  }, [fetchedModels, selectedModels]);
   973→
   974→  const handleFetchModels = useCallback(async () => {
   975→    if (!provider) return;
   976→    setFetching(true);
   977→    setFetchError('');
   978→    try {
   979→      const res = await fetch(`/api/providers/${provider.id}/fetch-models`, {
   980→        method: 'POST',
   981→      });
   982→      if (!res.ok) {
   983→        const body = await res.json().catch(() => ({}));
   984→        throw new Error(body.error ?? 'Failed to fetch models');
   985→      }
   986→      const data = await res.json();
   987→      const models: string[] = data.models;
   988→      if (models.length === 0) {
   989→        setFetchError('No models found. The API responded but returned an empty list.');
   990→      } else {
   991→        setFetchedModels(models);
   992→        // Auto-select all fetched models
   993→        setSelectedModels((prev) => [...new Set([...prev, ...models])]);
   994→      }
   995→    } catch (err) {
   996→      setFetchError(err instanceof Error ? err.message : 'Failed to fetch models');
   997→    } finally {
   998→      setFetching(false);
   999→    }
  1000→  }, [provider]);
  1001→
  1002→  const toggleModel = useCallback((model: string) => {
  1003→    setSelectedModels((prev) =>
  1004→      prev.includes(model)
  1005→        ? prev.filter((m) => m !== model)
  1006→        : [...prev, model]
  1007→    );
  1008→  }, []);
  1009→
  1010→  const addCustomModel = useCallback(() => {
  1011→    const trimmed = customModel.trim();
  1012→    if (trimmed && !selectedModels.includes(trimmed)) {
  1013→      setSelectedModels((prev) => [...prev, trimmed]);
  1014→      setCustomModel('');
  1015→    }
  1016→  }, [customModel, selectedModels]);
  1017→
  1018→  const selectAll = useCallback(() => {
  1019→    setSelectedModels(allModels);
  1020→  }, [allModels]);
  1021→
  1022→  const deselectAll = useCallback(() => {
  1023→    setSelectedModels([]);
  1024→  }, []);
  1025→
  1026→  const handleSave = useCallback(async () => {
  1027→    if (!provider) return;
  1028→    setSaving(true);
  1029→    try {
  1030→      await onUpdate(provider.id, { models: selectedModels });
  1031→      onOpenChange(false);
  1032→      toast.success('Models updated', {
  1033→        description: `${selectedModels.length} model${selectedModels.length !== 1 ? 's' : ''} configured for ${provider.name}.`,
  1034→      });
  1035→    } catch (err) {
  1036→      toast.error('Failed to save models', {
  1037→        description: err instanceof Error ? err.message : 'Unknown error',
  1038→      });
  1039→    } finally {
  1040→      setSaving(false);
  1041→    }
  1042→  }, [provider, selectedModels, onUpdate, onOpenChange]);
  1043→
  1044→  if (!provider) return null;
  1045→  const isLocal = provider.isLocal;
  1046→
  1047→  return (
  1048→    <Dialog open={open} onOpenChange={onOpenChange}>
  1049→      <DialogContent className="bg-popover border-border text-foreground sm:max-w-lg max-h-[85vh] flex flex-col">
  1050→        <DialogHeader>
  1051→          <DialogTitle className="text-popover-foreground">Configure Models</DialogTitle>
  1052→          <DialogDescription className="text-muted-foreground">
  1053→            Fetch and select available models for <span className="text-secondary-foreground font-medium">{provider.name}</span>.
  1054→          </DialogDescription>
  1055→        </DialogHeader>
  1056→
  1057→        <div className="flex-1 overflow-y-auto space-y-4 py-1">
  1058→          {/* Fetch button */}
  1059→          <Button
  1060→            type="button"
  1061→            variant="outline"
  1062→            className="w-full border-dashed border-input hover:border-violet-500/50 text-secondary-foreground hover:text-violet-300 hover:bg-violet-500/5 gap-2 h-10"
  1063→            onClick={handleFetchModels}
  1064→            disabled={fetching}
  1065→          >
  1066→            {fetching ? (
  1067→              <>
  1068→                <Loader2 className="h-4 w-4 animate-spin" />
  1069→                Fetching models…
  1070→              </>
  1071→            ) : (
  1072→              <>
  1073→                <Download className="h-4 w-4" />
  1074→                {fetchedModels.length > 0 ? 'Re-fetch Models from API' : 'Fetch Available Models from API'}
  1075→              </>
  1076→            )}
  1077→          </Button>
  1078→
  1079→          {fetchError && (
  1080→            <div className="rounded-lg border border-red-500/20 bg-red-500/5 p-3">
  1081→              <p className="text-xs text-red-400">{fetchError}</p>
  1082→              <p className="text-[11px] text-red-400/60 mt-1">
  1083→                {isLocal
  1084→                  ? 'Make sure your local server is running and accessible at the configured URL.'
  1085→                  : 'Check your API key and base URL. Some providers may require specific permissions.'}
  1086→              </p>
  1087→            </div>
  1088→          )}
  1089→
  1090→          {fetchedModels.length > 0 && (
  1091→            <p className="text-[11px] text-emerald-400/80">
  1092→              <Check className="h-3 w-3 inline mr-1" />
  1093→              Fetched {fetchedModels.length} model{fetchedModels.length !== 1 ? 's' : ''} from the API
  1094→            </p>
  1095→          )}
  1096→
  1097→          {/* Model list */}
  1098→          {allModels.length > 0 && (
  1099→            <>
  1100→              <div className="flex items-center justify-between">
  1101→                <Label className="text-secondary-foreground text-sm">
  1102→                  Models ({selectedModels.length} selected)
  1103→                </Label>
  1104→                <div className="flex gap-1">
  1105→                  <Button
  1106→                    type="button"
  1107→                    variant="ghost"
  1108→                    size="sm"
  1109→                    className="h-6 text-[11px] text-muted-foreground/70 hover:text-violet-400 hover:bg-violet-500/10 px-2"
  1110→                    onClick={selectAll}
  1111→                  >
  1112→                    Select all
  1113→                  </Button>
  1114→                  <Button
  1115→                    type="button"
  1116→                    variant="ghost"
  1117→                    size="sm"
  1118→                    className="h-6 text-[11px] text-muted-foreground/70 hover:text-muted-foreground hover:"
  1119→                    onClick={deselectAll}
  1120→                  >
  1121→                    Clear
  1122→                  </Button>
  1123→                </div>
  1124→              </div>
  1125→

  1400→export function ProviderManager() {
  1401→  const {
  1402→    providers,
  1403→    loading,
  1404→    error,
  1405→    fetchProviders,
  1406→    addProvider,
  1407→    updateProvider,
  1408→    deleteProvider,
  1409→    testProvider,
  1410→  } = useProviderStore();
  1411→
  1412→  // Dialog states
  1413→  const [formOpen, setFormOpen] = useState(false);
  1414→  const [editingProvider, setEditingProvider] = useState<LlmProvider | null>(null);
  1415→  const [deleteTarget, setDeleteTarget] = useState<LlmProvider | null>(null);
  1416→  const [deleting, setDeleting] = useState(false);
  1417→  const [testingId, setTestingId] = useState<string | null>(null);
  1418→  const [modelConfigTarget, setModelConfigTarget] = useState<LlmProvider | null>(null);
  1419→
  1420→  // Discovery state
  1421→  const [discovering, setDiscovering] = useState(false);
  1422→  const [discovered, setDiscovered] = useState<DiscoveredProvider[]>([]);
  1423→
  1424→  // Fetch providers on mount
  1425→  useEffect(() => {
  1426→    fetchProviders();
  1427→  }, [fetchProviders]);
  1428→
  1429→  // Show error toast
  1430→  useEffect(() => {
  1431→    if (error) {
  1432→      toast.error('Provider error', { description: error });
  1433→    }
  1434→  }, [error]);
  1435→
  1436→  // ---- Handlers ----
  1437→
  1438→  const handleOpenAdd = useCallback(() => {
  1439→    setEditingProvider(null);
  1440→    setFormOpen(true);
  1441→  }, []);
  1442→
  1443→  const handleOpenEdit = useCallback((provider: LlmProvider) => {
  1444→    setEditingProvider(provider);
  1445→    setFormOpen(true);
  1446→  }, []);
  1447→
  1448→  const handleSave = useCallback(
  1449→    async (data: ProviderFormData) => {
  1450→      const typeDef = PROVIDER_TYPES.find((t) => t.value === data.type);
  1451→
  1452→      if (editingProvider) {
  1453→        await updateProvider(editingProvider.id, {
  1454→          name: data.name,
  1455→          type: data.type,
  1456→          baseUrl: data.baseUrl,
  1457→          apiKey: data.apiKey,
  1458→          isLocal: typeDef?.isLocal ?? false,
  1459→          isActive: data.isActive,
  1460→          models: data.models,
  1461→        });
  1462→      } else {
  1463→        await addProvider({
  1464→          name: data.name,
  1465→          type: data.type,
  1466→          baseUrl: data.baseUrl,
  1467→          apiKey: data.apiKey,
  1468→          isLocal: typeDef?.isLocal ?? false,
  1469→          isActive: data.isActive,
  1470→          models: data.models,
  1471→        });
  1472→      }
  1473→    },
  1474→    [editingProvider, addProvider, updateProvider]
  1475→  );
  1476→
  1477→  const handleDelete = useCallback(async () => {
  1478→    if (!deleteTarget) return;
  1479→    setDeleting(true);
  1480→    try {
  1481→      await deleteProvider(deleteTarget.id);
  1482→      toast.success('Provider deleted', {
  1483→        description: `${deleteTarget.name} has been removed.`,
  1484→      });
  1485→      setDeleteTarget(null);
  1486→    } catch (err) {
  1487→      toast.error('Failed to delete provider', {
  1488→        description: err instanceof Error ? err.message : 'Unknown error',
  1489→      });
  1490→    } finally {
  1491→      setDeleting(false);
  1492→    }
  1493→  }, [deleteTarget, deleteProvider]);
  1494→
  1495→  const handleTest = useCallback(
  1496→    async (id: string) => {
  1497→      setTestingId(id);
  1498→      try {
  1499→        const success = await testProvider(id);
  1500→        const provider = providers.find((p) => p.id === id);
  1501→        if (success) {
  1502→          toast.success('Connection successful', {
  1503→            description: `${provider?.name ?? 'Provider'} is reachable and responding.`,
  1504→          });
  1505→        } else {
  1506→          toast.error('Connection failed', {
  1507→            description: `Could not reach ${provider?.name ?? 'provider'}. Check the URL and API key.`,
  1508→          });
  1509→        }
  1510→      } catch {
  1511→        toast.error('Connection test error', {
  1512→          description: 'An unexpected error occurred during the test.',
  1513→        });
  1514→      } finally {
  1515→        setTestingId(null);
  1516→      }
  1517→    },
  1518→    [testProvider, providers]
  1519→  );
  1520→
  1521→  const handleDiscover = useCallback(async () => {
  1522→    setDiscovering(true);
  1523→    setDiscovered([]);
  1524→    try {
  1525→      const res = await fetch('/api/discover');
  1526→      if (!res.ok) {
  1527→        throw new Error('Discovery request failed');
  1528→      }
  1529→      const results: DiscoveredProvider[] = await res.json();
  1530→      setDiscovered(results);
  1531→
  1532→      if (results.length === 0) {
  1533→        toast.info('No local servers found', {
  1534→          description:
  1535→            'Make sure Ollama, LM Studio, vLLM, or another local server is running on its default port.',
  1536→        });
  1537→      } else {
  1538→        toast.success(`Found ${results.length} local ${results.length === 1 ? 'server' : 'servers'}`, {
  1539→          description: 'Select which ones you\'d like to add as providers.',
  1540→        });
  1541→      }
  1542→    } catch (err) {
  1543→      toast.error('Discovery failed', {
  1544→        description: err instanceof Error ? err.message : 'Could not scan for local servers.',
  1545→      });
  1546→    } finally {
  1547→      setDiscovering(false);
  1548→    }
  1549→  }, []);
  1550→
  1551→  const handleAddDiscovered = useCallback(
  1552→    (result: DiscoveredProvider) => {
  1553→      const typeDef = PROVIDER_TYPES.find((t) => t.value === result.type);
  1554→      addProvider({
  1555→        name: `${result.name} (Local)`,
  1556→        type: result.type as ProviderType,
  1557→        baseUrl: result.baseUrl,
  1558→        apiKey: '',
  1559→        isLocal: true,
  1560→        isActive: true,
  1561→        models: result.models,
  1562→      }).then(() => {
  1563→        setDiscovered((prev) => prev.filter((r) => r.port !== result.port));
  1564→        toast.success(`${result.name} added`, {
  1565→          description: `Added with ${result.models.length} model(s) on port ${result.port}.`,
  1566→        });
  1567→      }).catch((err) => {
  1568→        toast.error('Failed to add discovered provider', {
  1569→          description: err instanceof Error ? err.message : 'Unknown error',
  1570→        });
  1571→      });
  1572→    },
  1573→    [addProvider]
  1574→  );
  1575→
  1576→  // ---- Render ----
  1577→
  1578→  return (
  1579→    <div className="h-full overflow-y-auto">
  1580→      <div className="max-w-6xl mx-auto p-6 space-y-6">
  1581→        {/* Header */}
  1582→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
  1583→          <div className="flex items-center gap-3">
  1584→            <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-violet-500/10 border border-violet-500/20">
  1585→              <Server className="h-5 w-5 text-violet-400" />
  1586→            </div>
  1587→            <div>
  1588→              <h1 className="text-xl font-bold text-foreground tracking-tight">Model Gateway</h1>
  1589→              <p className="text-sm text-muted-foreground mt-0.5">
  1590→                Configure LLM providers for your agents
  1591→              </p>
  1592→            </div>
  1593→          </div>
  1594→
  1595→          <div className="flex items-center gap-2 shrink-0">
  1596→            <Button
  1597→              variant="outline"
  1598→              onClick={handleDiscover}
  1599→              disabled={discovering}
  1600→              className="border-input text-secondary-foreground hover:"
  1601→            >
  1602→              {discovering ? (
  1603→                <Loader2 className="h-4 w-4 animate-spin" />
  1604→              ) : (
  1605→                <Wifi className="h-4 w-4" />
  1606→              )}
  1607→              Discover Local
  1608→            </Button>
  1609→            <Button
  1610→              onClick={handleOpenAdd}
  1611→              className="bg-violet-600 hover:bg-violet-500 text-white gap-2"
  1612→            >
  1613→              <Plus className="h-4 w-4" />
  1614→              Add Provider
  1615→            </Button>
  1616→          </div>
  1617→        </div>
  1618→
  1619→        {/* Discovery results */}
  1620→        <AnimatePresence>
  1621→          {discovered.length > 0 && (
  1622→            <DiscoveryResults
  1623→              results={discovered}
  1624→              onAdd={handleAddDiscovered}
  1625→              onDismiss={() => setDiscovered([])}
  1626→            />
  1627→          )}
  1628→        </AnimatePresence>
  1629→
  1630→        {/* Content */}
  1631→        {loading && providers.length === 0 ? (
  1632→          <ProviderSkeletonGrid />
  1633→        ) : providers.length === 0 ? (
  1634→          <EmptyState onAdd={handleOpenAdd} />

  1636→          <AnimatePresence mode="popLayout">

  1638→              {providers.map((provider, i) => (
  1639→                <ProviderCard
  1640→                  key={provider.id}
  1641→                  provider={provider}

  1643→                  onEdit={handleOpenEdit}
  1644→                  onTest={handleTest}
  1645→                  onDelete={setDeleteTarget}

  1647→                  testingId={testingId}

  1655→      {/* Form Dialog */}
  1656→      <ProviderFormDialog
