   113→   113→  // ---------------------------------------------------------------------------

   116→   116→
   117→   117→  const fetchSettings = useCallback(async () => {
   118→   118→    setLoading(true);
   119→   119→    try {
   120→   120→      const res = await fetch('/api/settings');
   121→   121→      if (res.ok) {
   122→   122→        const data = await res.json();
   123→   123→        setSettings({ ...DEFAULT_SETTINGS, ...data });
   124→   124→      }
   125→   125→    } catch {
   126→   126→      // use defaults
   127→   127→    } finally {
   128→   128→      setLoading(false);
   129→   129→    }
   130→   130→  }, []);
   131→   131→
   132→   132→  useEffect(() => {
   133→   133→    fetchSettings();
   134→   134→  }, [fetchSettings]);
   135→   135→
   136→   136→  // ---------------------------------------------------------------------------
   137→   137→  // Save settings (debounced)
   138→   138→  // ---------------------------------------------------------------------------
   139→   139→
   140→   140→  const saveSettings = useCallback(async (newSettings: AppSettings) => {
   141→   141→    setSaving(true);
   142→   142→    try {
   143→   143→      const res = await fetch('/api/settings', {
   144→   144→        method: 'PATCH',
   145→   145→        headers: { 'Content-Type': 'application/json' },
   146→   146→        body: JSON.stringify(newSettings),
   147→   147→      });

   149→   149→    } catch {
   150→   150→      toast.error('Failed to save settings');
   151→   151→    } finally {
   152→   152→      setSaving(false);
   153→   153→    }
   154→   154→  }, []);
   155→   155→
   156→   156→  const updateSetting = useCallback(
   157→   157→    <K extends keyof AppSettings>(key: K, value: AppSettings[K]) => {
   158→   158→      setSettings((prev) => {
   159→   159→        const updated = { ...prev, [key]: value };
   160→   160→        // Debounced save
   161→   161→        if (debounceRef.current) clearTimeout(debounceRef.current);
   162→   162→        debounceRef.current = setTimeout(() => {
   163→   163→          saveSettings(updated);
   164→   164→        }, 500);
   165→   165→        return updated;
   166→   166→      });
   167→   167→    },
   168→   168→    [saveSettings],
   169→   169→  );

   171→   171→  const updateAgentBudget = useCallback(

   174→   174→        const updated = {
   175→   175→          ...prev,
   176→   176→          agentBudgetLimits: { ...prev.agentBudgetLimits, [agentId]: value },
   177→   177→        };

   179→   179→        debounceRef.current = setTimeout(() => {
   180→   180→          saveSettings(updated);
   181→   181→        }, 500);
   182→   182→        return updated;
   183→   183→      });
   184→   184→    },
   185→   185→    [saveSettings],
   186→   186→  );
   187→   187→

   193→   193→  const handleResetCosts = useCallback(() => {

   215→   215→  };

   229→   229→                <Settings className="h-4.5 w-4.5 text-violet-400" />
   230→   230→              </div>

   247→   247→                  Appearance, language, and default behavior

   299→   299→                  >

   301→   301→                      <SelectValue />

   356→   356→                      Budget limits and spending alerts

   358→   358→                  </div>
