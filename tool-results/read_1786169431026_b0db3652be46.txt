   431→// =============================================================================
   432→// Provider Form Dialog
   433→// =============================================================================
   434→
   435→function ProviderFormDialog({
   436→  open,
   437→  onOpenChange,
   438→  editingProvider,
   439→  onSave,
   440→}: {
   441→  open: boolean;
   442→  onOpenChange: (open: boolean) => void;
   443→  editingProvider: LlmProvider | null;
   444→  onSave: (data: ProviderFormData) => Promise<void>;
   445→}) {
   446→  const [form, setForm] = useState<ProviderFormData>(EMPTY_FORM);
   447→  const [showApiKey, setShowApiKey] = useState(false);
   448→  const [saving, setSaving] = useState(false);
   449→  const [customModel, setCustomModel] = useState('');
   450→
   451→  const isEditing = editingProvider !== null;
   452→  const isLocal = PROVIDER_TYPES.find((t) => t.value === form.type)?.isLocal ?? false;
   453→  const defaultModels = useMemo(
   454→    () => PROVIDER_TYPES.find((t) => t.value === form.type)?.models ?? [],
   455→    [form.type]
   456→  );
   457→
   458→  // Populate form when editing or resetting
   459→  useEffect(() => {
   460→    if (editingProvider) {
   461→      setForm({
   462→        name: editingProvider.name,
   463→        type: editingProvider.type,
   464→        baseUrl: editingProvider.baseUrl,
   465→        apiKey: editingProvider.apiKey,
   466→        models: editingProvider.models,
   467→        isActive: editingProvider.isActive,
   468→      });
   469→    } else {
   470→      setForm(EMPTY_FORM);
   471→    }
   472→    setShowApiKey(false);
   473→    setCustomModel('');
   474→  }, [editingProvider, open]);
   475→
   476→  // Auto-fill base URL when type changes
   477→  const handleTypeChange = useCallback(
   478→    (value: string) => {
   479→      const type = value as ProviderType;
   480→      const def = PROVIDER_TYPES.find((t) => t.value === type);
   481→      setForm((prev) => ({
   482→        ...prev,
   483→        type,
   484→        baseUrl: def?.defaultBaseUrl ?? '',
   485→        models: [],
   486→      }));
   487→    },
   488→    []
   489→  );
   490→
   491→  const toggleModel = useCallback((model: string) => {
   492→    setForm((prev) => ({
   493→      ...prev,
   494→      models: prev.models.includes(model)
   495→        ? prev.models.filter((m) => m !== model)
   496→        : [...prev.models, model],
   497→    }));
   498→  }, []);
   499→
   500→  const addCustomModel = useCallback(() => {
   501→    const trimmed = customModel.trim();
   502→    if (trimmed && !form.models.includes(trimmed)) {
   503→      setForm((prev) => ({
   504→        ...prev,
   505→        models: [...prev.models, trimmed],
   506→      }));
   507→      setCustomModel('');
   508→    }
   509→  }, [customModel, form.models]);
   510→
   511→  const removeModel = useCallback((model: string) => {
   512→    setForm((prev) => ({
   513→      ...prev,
   514→      models: prev.models.filter((m) => m !== model),
   515→    }));
   516→  }, []);
   517→
   518→  const handleSave = useCallback(async () => {
   519→    if (!form.name.trim()) {
   520→      toast.error('Provider name is required');
   521→      return;
   522→    }
   523→    if (!form.baseUrl.trim()) {
   524→      toast.error('Base URL is required');
   525→      return;
   526→    }
   527→    if (!isLocal && !form.apiKey.trim() && form.type !== 'custom') {
   528→      toast.error('API key is required for cloud providers');
   529→      return;
   530→    }
   531→
   532→    setSaving(true);
   533→    try {
   534→      await onSave(form);
   535→      onOpenChange(false);
   536→      toast.success(isEditing ? 'Provider updated' : 'Provider added', {
   537→        description: `${form.name} has been ${isEditing ? 'updated' : 'added'} successfully.`,
   538→      });
   539→    } catch (err) {
   540→      toast.error(isEditing ? 'Failed to update provider' : 'Failed to add provider', {
   541→        description: err instanceof Error ? err.message : 'Unknown error',
   542→      });
   543→    } finally {
   544→      setSaving(false);
   545→    }
   546→  }, [form, isLocal, isEditing, onSave, onOpenChange]);
   547→
   548→  const selectAllDefaults = useCallback(() => {
   549→    setForm((prev) => ({
   550→      ...prev,
   551→      models: [...new Set([...prev.models, ...defaultModels])],
   552→    }));
   553→  }, [defaultModels]);
   554→
   555→  const deselectAllDefaults = useCallback(() => {
   556→    setForm((prev) => ({
   557→      ...prev,
   558→      models: prev.models.filter((m) => !defaultModels.includes(m)),
   559→    }));
   560→  }, [defaultModels]);
   561→
   562→  return (
   563→    <Dialog open={open} onOpenChange={onOpenChange}>
   564→      <DialogContent className="bg-zinc-900 border-zinc-800 text-zinc-100 sm:max-w-lg max-h-[90vh] flex flex-col">
   565→        <DialogHeader>
   566→          <DialogTitle className="text-zinc-100">
   567→            {isEditing ? 'Edit Provider' : 'Add Provider'}
   568→          </DialogTitle>
   569→          <DialogDescription className="text-zinc-400">
   570→            {isEditing
   571→              ? 'Update the configuration for this LLM provider.'

   997→export function ProviderManager() {
   998→  const {
   999→    providers,
  1000→    loading,
  1001→    error,
  1002→    fetchProviders,
  1003→    addProvider,
  1004→    updateProvider,
  1005→    deleteProvider,
  1006→    testProvider,
  1007→  } = useProviderStore();
  1008→
  1009→  // Dialog states
  1010→  const [formOpen, setFormOpen] = useState(false);
  1011→  const [editingProvider, setEditingProvider] = useState<LlmProvider | null>(null);
  1012→  const [deleteTarget, setDeleteTarget] = useState<LlmProvider | null>(null);
  1013→  const [deleting, setDeleting] = useState(false);
  1014→  const [testingId, setTestingId] = useState<string | null>(null);
  1015→
  1016→  // Discovery state
  1017→  const [discovering, setDiscovering] = useState(false);
  1018→  const [discovered, setDiscovered] = useState<DiscoveredProvider[]>([]);
  1019→
  1020→  // Fetch providers on mount
  1021→  useEffect(() => {
  1022→    fetchProviders();
  1023→  }, [fetchProviders]);
  1024→
  1025→  // Show error toast
  1026→  useEffect(() => {
  1027→    if (error) {
  1028→      toast.error('Provider error', { description: error });
  1029→    }
  1030→  }, [error]);
  1031→
  1032→  // ---- Handlers ----
  1033→
  1034→  const handleOpenAdd = useCallback(() => {
  1035→    setEditingProvider(null);
  1036→    setFormOpen(true);
  1037→  }, []);
  1038→
  1039→  const handleOpenEdit = useCallback((provider: LlmProvider) => {
  1040→    setEditingProvider(provider);
  1041→    setFormOpen(true);
  1042→  }, []);
  1043→
  1044→  const handleSave = useCallback(
  1045→    async (data: ProviderFormData) => {
  1046→      const typeDef = PROVIDER_TYPES.find((t) => t.value === data.type);
  1047→
  1048→      if (editingProvider) {
  1049→        await updateProvider(editingProvider.id, {
  1050→          name: data.name,
  1051→          type: data.type,
  1052→          baseUrl: data.baseUrl,
  1053→          apiKey: data.apiKey,
  1054→          isLocal: typeDef?.isLocal ?? false,
  1055→          isActive: data.isActive,
  1056→          models: data.models,
  1057→        });
  1058→      } else {
  1059→        await addProvider({
  1060→          name: data.name,
  1061→          type: data.type,
  1062→          baseUrl: data.baseUrl,
  1063→          apiKey: data.apiKey,
  1064→          isLocal: typeDef?.isLocal ?? false,
  1065→          isActive: data.isActive,
  1066→          models: data.models,
  1067→        });
  1068→      }
  1069→    },
  1070→    [editingProvider, addProvider, updateProvider]
  1071→  );
  1072→
  1073→  const handleDelete = useCallback(async () => {
  1074→    if (!deleteTarget) return;
  1075→    setDeleting(true);
  1076→    try {
  1077→      await deleteProvider(deleteTarget.id);
  1078→      toast.success('Provider deleted', {
  1079→        description: `${deleteTarget.name} has been removed.`,
  1080→      });
  1081→      setDeleteTarget(null);
  1082→    } catch (err) {
  1083→      toast.error('Failed to delete provider', {
  1084→        description: err instanceof Error ? err.message : 'Unknown error',
  1085→      });
  1086→    } finally {
  1087→      setDeleting(false);
  1088→    }
  1089→  }, [deleteTarget, deleteProvider]);
  1090→
  1091→  const handleTest = useCallback(
  1092→    async (id: string) => {
  1093→      setTestingId(id);
  1094→      try {
  1095→        const success = await testProvider(id);
  1096→        const provider = providers.find((p) => p.id === id);
  1097→        if (success) {
  1098→          toast.success('Connection successful', {
  1099→            description: `${provider?.name ?? 'Provider'} is reachable and responding.`,
  1100→          });
  1101→        } else {
  1102→          toast.error('Connection failed', {
  1103→            description: `Could not reach ${provider?.name ?? 'provider'}. Check the URL and API key.`,
  1104→          });
  1105→        }
  1106→      } catch {
  1107→        toast.error('Connection test error', {
  1108→          description: 'An unexpected error occurred during the test.',
  1109→        });
  1110→      } finally {
  1111→        setTestingId(null);
  1112→      }
  1113→    },
  1114→    [testProvider, providers]
  1115→  );
  1116→
  1117→  const handleDiscover = useCallback(async () => {
  1118→    setDiscovering(true);
  1119→    setDiscovered([]);
  1120→    try {
  1121→      const res = await fetch('/api/discover');
  1122→      if (!res.ok) {
  1123→        throw new Error('Discovery request failed');
  1124→      }
  1125→      const results: DiscoveredProvider[] = await res.json();
  1126→      setDiscovered(results);
  1127→
  1128→      if (results.length === 0) {
  1129→        toast.info('No local servers found', {
  1130→          description:
  1131→            'Make sure Ollama, LM Studio, vLLM, or another local server is running on its default port.',
  1132→        });
  1133→      } else {
