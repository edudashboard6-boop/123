   460→// Provider Form Dialog — Step 1: Connection details only (models configured later)

   462→
   463→function ProviderFormDialog({
   464→  open,
   465→  onOpenChange,
   466→  editingProvider,
   467→  onSave,
   468→}: {
   469→  open: boolean;
   470→  onOpenChange: (open: boolean) => void;
   471→  editingProvider: LlmProvider | null;
   472→  onSave: (data: ProviderFormData) => Promise<void>;
   473→}) {
   474→  const [form, setForm] = useState<ProviderFormData>(EMPTY_FORM);
   475→  const [showApiKey, setShowApiKey] = useState(false);
   476→  const [saving, setSaving] = useState(false);
   477→
   478→  const isEditing = editingProvider !== null;
   479→  const isLocal = PROVIDER_TYPES.find((t) => t.value === form.type)?.isLocal ?? false;
   480→
   481→  // Populate form when editing or resetting
   482→  useEffect(() => {
   483→    if (editingProvider) {
   484→      setForm({
   485→        name: editingProvider.name,
   486→        type: editingProvider.type,
   487→        baseUrl: editingProvider.baseUrl,
   488→        apiKey: editingProvider.apiKey,
   489→        models: editingProvider.models,
   490→        isActive: editingProvider.isActive,
   491→      });
   492→    } else {
   493→      setForm(EMPTY_FORM);
   494→    }
   495→    setShowApiKey(false);
   496→  }, [editingProvider, open]);
   497→
   498→  // Auto-fill base URL when type changes
   499→  const handleTypeChange = useCallback(
   500→    (value: string) => {
   501→      const type = value as ProviderType;
   502→      const def = PROVIDER_TYPES.find((t) => t.value === type);
   503→      setForm((prev) => ({
   504→        ...prev,
   505→        type,
   506→        baseUrl: def?.defaultBaseUrl ?? '',
   507→      }));
   508→    },
   509→    []
   510→  );
   511→
   512→  const handleSave = useCallback(async () => {
   513→    if (!form.name.trim()) {
   514→      toast.error('Provider name is required');
   515→      return;
   516→    }
   517→    if (!form.baseUrl.trim()) {
   518→      toast.error('Base URL is required');
   519→      return;
   520→    }
   521→    if (!isLocal && !form.apiKey.trim() && form.type !== 'custom') {
   522→      toast.error('API key is required for cloud providers');
   523→      return;
   524→    }
   525→
   526→    setSaving(true);
   527→    try {
   528→      await onSave(form);
   529→      onOpenChange(false);
   530→      toast.success(isEditing ? 'Provider updated' : 'Provider added', {
   531→        description: `${form.name} has been ${isEditing ? 'updated' : 'added'}. ${!isEditing ? 'Now configure its models.' : ''}`,
   532→      });
   533→    } catch (err) {
   534→      toast.error(isEditing ? 'Failed to update provider' : 'Failed to add provider', {
   535→        description: err instanceof Error ? err.message : 'Unknown error',
   536→      });
   537→    } finally {
   538→      setSaving(false);
   539→    }
   540→  }, [form, isLocal, isEditing, onSave, onOpenChange]);
   541→
   542→  return (
   543→    <Dialog open={open} onOpenChange={onOpenChange}>
   544→      <DialogContent className="bg-zinc-900 border-zinc-800 text-zinc-100 sm:max-w-lg">
   545→        <DialogHeader>
   546→          <DialogTitle className="text-zinc-100">
   547→            {isEditing ? 'Edit Provider' : 'Add Provider'}
   548→          </DialogTitle>
   549→          <DialogDescription className="text-zinc-400">
   550→            {isEditing
   551→              ? 'Update the connection settings for this provider.'
   552→              : 'Configure a new LLM provider. You\'ll select models after saving.'}
   553→          </DialogDescription>
   554→        </DialogHeader>
   555→
   556→        <div className="space-y-5 py-2">
   557→          {/* Name */}
   558→          <div className="space-y-2">
   559→            <Label htmlFor="provider-name" className="text-zinc-300 text-sm">
   560→              Name
   561→            </Label>
   562→            <Input
   563→              id="provider-name"
   564→              placeholder="e.g. My GPT-4o Instance"
   565→              value={form.name}
   566→              onChange={(e) => setForm((p) => ({ ...p, name: e.target.value }))}
   567→              className="bg-zinc-800 border-zinc-700 text-zinc-100 placeholder:text-zinc-500 focus-visible:ring-violet-500/50 focus-visible:border-violet-500/50"
   568→            />
   569→          </div>
   570→
   571→          {/* Provider Type */}
   572→          <div className="space-y-2">
   573→            <Label className="text-zinc-300 text-sm">Provider Type</Label>
   574→            <Select value={form.type} onValueChange={handleTypeChange}>
   575→              <SelectTrigger className="bg-zinc-800 border-zinc-700 text-zinc-100 focus:ring-violet-500/50 focus:border-violet-500/50">
   576→                <SelectValue placeholder="Select a provider type" />
   577→              </SelectTrigger>
   578→              <SelectContent className="bg-zinc-900 border-zinc-700">
   579→                <SelectGroup>
   580→                  <SelectLabel className="text-zinc-500 text-xs">Cloud Providers</SelectLabel>
   581→                  {PROVIDER_TYPES.filter((t) => !t.isLocal).map((t) => (
   582→                    <SelectItem key={t.value} value={t.value} className="text-zinc-200 focus:bg-zinc-800 focus:text-zinc-100">
   583→                      {t.label}
   584→                    </SelectItem>
   585→                  ))}
   586→                </SelectGroup>
   587→                <SelectGroup>
   588→                  <SelectLabel className="text-zinc-500 text-xs">Local Providers</SelectLabel>
   589→                  {PROVIDER_TYPES.filter((t) => t.isLocal).map((t) => (
   590→                    <SelectItem key={t.value} value={t.value} className="text-zinc-200 focus:bg-zinc-800 focus:text-zinc-100">
   591→                      {t.label}
   592→                    </SelectItem>
   593→                  ))}
   594→                </SelectGroup>
   595→              </SelectContent>
   596→            </Select>
   597→          </div>
   598→

   711→function ModelConfigDialog({
   712→  open,
   713→  onOpenChange,
   714→  provider,
   715→  onUpdate,
   716→}: {
   717→  open: boolean;
   718→  onOpenChange: (open: boolean) => void;
   719→  provider: LlmProvider | null;
   720→  onUpdate: (id: string, data: { models: string[] }) => Promise<void>;
   721→}) {
   722→  const [selectedModels, setSelectedModels] = useState<string[]>([]);
   723→  const [fetchedModels, setFetchedModels] = useState<string[]>([]);
   724→  const [customModel, setCustomModel] = useState('');
   725→  const [fetching, setFetching] = useState(false);
   726→  const [saving, setSaving] = useState(false);
   727→  const [fetchError, setFetchError] = useState('');
   728→
   729→  // Sync selected models with provider when dialog opens
   730→  useEffect(() => {
   731→    if (provider) {
   732→      setSelectedModels(provider.models);
   733→      setFetchedModels([]);
   734→      setFetchError('');
   735→      setCustomModel('');
   736→    }
   737→  }, [provider, open]);
   738→
   739→  const allModels = useMemo(() => {
   740→    const combined = [...new Set([...fetchedModels, ...selectedModels])];
   741→    return combined.sort();
   742→  }, [fetchedModels, selectedModels]);
   743→
   744→  const handleFetchModels = useCallback(async () => {
   745→    if (!provider) return;
   746→    setFetching(true);
   747→    setFetchError('');
   748→    try {
   749→      const res = await fetch(`/api/providers/${provider.id}/fetch-models`, {
   750→        method: 'POST',
   751→      });
   752→      if (!res.ok) {
   753→        const body = await res.json().catch(() => ({}));
   754→        throw new Error(body.error ?? 'Failed to fetch models');
   755→      }
   756→      const data = await res.json();
   757→      const models: string[] = data.models;
   758→      if (models.length === 0) {
   759→        setFetchError('No models found. The API responded but returned an empty list.');
   760→      } else {
   761→        setFetchedModels(models);
   762→        // Auto-select all fetched models
   763→        setSelectedModels((prev) => [...new Set([...prev, ...models])]);
   764→      }
   765→    } catch (err) {
   766→      setFetchError(err instanceof Error ? err.message : 'Failed to fetch models');
   767→    } finally {
   768→      setFetching(false);
   769→    }
   770→  }, [provider]);
   771→
   772→  const toggleModel = useCallback((model: string) => {
   773→    setSelectedModels((prev) =>
   774→      prev.includes(model)
   775→        ? prev.filter((m) => m !== model)
   776→        : [...prev, model]
   777→    );
   778→  }, []);
   779→
   780→  const addCustomModel = useCallback(() => {
   781→    const trimmed = customModel.trim();
   782→    if (trimmed && !selectedModels.includes(trimmed)) {
   783→      setSelectedModels((prev) => [...prev, trimmed]);
   784→      setCustomModel('');
   785→    }
   786→  }, [customModel, selectedModels]);
   787→
   788→  const selectAll = useCallback(() => {
   789→    setSelectedModels(allModels);
   790→  }, [allModels]);
   791→
   792→  const deselectAll = useCallback(() => {
   793→    setSelectedModels([]);
   794→  }, []);
   795→
   796→  const handleSave = useCallback(async () => {
   797→    if (!provider) return;
   798→    setSaving(true);
   799→    try {
   800→      await onUpdate(provider.id, { models: selectedModels });
   801→      onOpenChange(false);
   802→      toast.success('Models updated', {
   803→        description: `${selectedModels.length} model${selectedModels.length !== 1 ? 's' : ''} configured for ${provider.name}.`,
   804→      });
   805→    } catch (err) {
   806→      toast.error('Failed to save models', {
   807→        description: err instanceof Error ? err.message : 'Unknown error',
   808→      });
   809→    } finally {
   810→      setSaving(false);
   811→    }
   812→  }, [provider, selectedModels, onUpdate, onOpenChange]);
   813→
   814→  if (!provider) return null;
   815→  const isLocal = provider.isLocal;
   816→
   817→  return (
   818→    <Dialog open={open} onOpenChange={onOpenChange}>
   819→      <DialogContent className="bg-zinc-900 border-zinc-800 text-zinc-100 sm:max-w-lg max-h-[85vh] flex flex-col">
   820→        <DialogHeader>
   821→          <DialogTitle className="text-zinc-100">Configure Models</DialogTitle>
   822→          <DialogDescription className="text-zinc-400">
   823→            Fetch and select available models for <span className="text-zinc-200 font-medium">{provider.name}</span>.
   824→          </DialogDescription>
   825→        </DialogHeader>
   826→
   827→        <div className="flex-1 overflow-y-auto space-y-4 py-1">
   828→          {/* Fetch button */}
   829→          <Button
   830→            type="button"
   831→            variant="outline"
   832→            className="w-full border-dashed border-zinc-700 hover:border-violet-500/50 text-zinc-300 hover:text-violet-300 hover:bg-violet-500/5 gap-2 h-10"
   833→            onClick={handleFetchModels}
   834→            disabled={fetching}
   835→          >
   836→            {fetching ? (
   837→              <>
   838→                <Loader2 className="h-4 w-4 animate-spin" />
   839→                Fetching models…
   840→              </>
   841→            ) : (
   842→              <>
   843→                <Download className="h-4 w-4" />
   844→                {fetchedModels.length > 0 ? 'Re-fetch Models from API' : 'Fetch Available Models from API'}
   845→              </>
   846→            )}
   847→          </Button>
   848→
   849→          {fetchError && (
   850→            <div className="rounded-lg border border-red-500/20 bg-red-500/5 p-3">
   851→              <p className="text-xs text-red-400">{fetchError}</p>
   852→              <p className="text-[11px] text-red-400/60 mt-1">
   853→                {isLocal
   854→                  ? 'Make sure your local server is running and accessible at the configured URL.'
   855→                  : 'Check your API key and base URL. Some providers may require specific permissions.'}
   856→              </p>
   857→            </div>
   858→          )}
   859→
   860→          {fetchedModels.length > 0 && (
   861→            <p className="text-[11px] text-emerald-400/80">
   862→              <Check className="h-3 w-3 inline mr-1" />
   863→              Fetched {fetchedModels.length} model{fetchedModels.length !== 1 ? 's' : ''} from the API
   864→            </p>
   865→          )}
   866→
   867→          {/* Model list */}
   868→          {allModels.length > 0 && (
   869→            <>
   870→              <div className="flex items-center justify-between">
   871→                <Label className="text-zinc-300 text-sm">
   872→                  Models ({selectedModels.length} selected)
   873→                </Label>
   874→                <div className="flex gap-1">
   875→                  <Button
   876→                    type="button"
   877→                    variant="ghost"
   878→                    size="sm"
   879→                    className="h-6 text-[11px] text-zinc-500 hover:text-violet-400 hover:bg-violet-500/10 px-2"
   880→                    onClick={selectAll}
   881→                  >
   882→                    Select all
   883→                  </Button>
   884→                  <Button
   885→                    type="button"
   886→                    variant="ghost"
   887→                    size="sm"
   888→                    className="h-6 text-[11px] text-zinc-500 hover:text-zinc-400 hover:bg-zinc-800 px-2"
   889→                    onClick={deselectAll}
   890→                  >
   891→                    Clear
   892→                  </Button>
   893→                </div>
   894→              </div>
   895→
   896→              <div className="grid grid-cols-1 gap-1 max-h-52 overflow-y-auto pr-1">
   897→                {allModels.map((model) => (
   898→                  <label
   899→                    key={model}
   900→                    className="flex items-center gap-2.5 rounded-md px-2.5 py-1.5 hover:bg-zinc-800/70 cursor-pointer transition-colors"
   901→                  >
   902→                    <Checkbox
   903→                      checked={selectedModels.includes(model)}
   904→                      onCheckedChange={() => toggleModel(model)}
   905→                      className="border-zinc-600 data-[state=checked]:bg-violet-600 data-[state=checked]:border-violet-600"
   906→                    />
   907→                    <span className="text-xs text-zinc-300 truncate">{model}</span>
   908→                  </label>
   909→                ))}
   910→              </div>
   911→            </>
   912→          )}
   913→
   914→          {/* Custom model input */}
   915→          <div className="space-y-2">
   916→            <Label className="text-zinc-400 text-xs">Or add a model manually:</Label>
   917→            <div className="flex gap-2">
   918→              <Input
   919→                placeholder="e.g. gpt-4o-2024-08-06"
   920→                value={customModel}
   921→                onChange={(e) => setCustomModel(e.target.value)}
   922→                onKeyDown={(e) => {
   923→                  if (e.key === 'Enter') {
   924→                    e.preventDefault();
   925→                    addCustomModel();
   926→                  }
   927→                }}
   928→                className="bg-zinc-800 border-zinc-700 text-zinc-100 placeholder:text-zinc-500 text-xs focus-visible:ring-violet-500/50 focus-visible:border-violet-500/50"
   929→              />
   930→              <Button
   931→                type="button"
   932→                variant="outline"
   933→                size="icon"
   934→                className="h-9 w-9 shrink-0 border-zinc-700 text-zinc-400 hover:text-zinc-100 hover:bg-zinc-800"
   935→                onClick={addCustomModel}
   936→              >
   937→                <Plus className="h-3.5 w-3.5" />
   938→              </Button>
   939→            </div>
   940→          </div>
   941→
   942→          {/* Currently selected as badges */}
   943→          {selectedModels.length > 0 && (
   944→            <div className="flex flex-wrap gap-1.5">
   945→              {selectedModels.map((model) => (
   946→                <Badge
   947→                  key={model}
   948→                  variant="secondary"
   949→                  className="text-[11px] px-2 py-0.5 h-6 bg-violet-500/15 text-violet-300 border-violet-500/25 hover:bg-violet-500/25 font-normal gap-1"
   950→                >
   951→                  {model.length > 24 ? model.slice(0, 24) + '…' : model}
   952→                  <button
   953→                    type="button"
   954→                    className="ml-0.5 hover:text-violet-100 transition-colors"
   955→                    onClick={() => toggleModel(model)}
   956→                  >
   957→                    <X className="h-2.5 w-2.5" />
   958→                  </button>
   959→                </Badge>
   960→              ))}
   961→            </div>
   962→          )}
   963→        </div>
   964→
   965→        <DialogFooter className="pt-4 border-t border-zinc-800">
   966→          <Button
   967→            variant="outline"
   968→            onClick={() => onOpenChange(false)}
   969→            className="border-zinc-700 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100"
   970→          >
   971→            Cancel
   972→          </Button>
   973→          <Button
   974→            onClick={handleSave}
   975→            disabled={saving || selectedModels.length === 0}
   976→            className="bg-violet-600 hover:bg-violet-500 text-white min-w-[100px]"
   977→          >
   978→            {saving ? (
   979→              <>
   980→                <Loader2 className="h-4 w-4 animate-spin mr-2" />
   981→                Saving…
   982→              </>
   983→            ) : (
   984→              `Save ${selectedModels.length} Model${selectedModels.length !== 1 ? 's' : ''}`
   985→            )}
   986→          </Button>
   987→        </DialogFooter>
   988→      </DialogContent>
   989→    </Dialog>
   990→  );
   991→}

  1170→export function ProviderManager() {
  1171→  const {
  1172→    providers,
  1173→    loading,
  1174→    error,
  1175→    fetchProviders,
  1176→    addProvider,
  1177→    updateProvider,
  1178→    deleteProvider,
  1179→    testProvider,
  1180→  } = useProviderStore();
  1181→
  1182→  // Dialog states
  1183→  const [formOpen, setFormOpen] = useState(false);
  1184→  const [editingProvider, setEditingProvider] = useState<LlmProvider | null>(null);
  1185→  const [deleteTarget, setDeleteTarget] = useState<LlmProvider | null>(null);
  1186→  const [deleting, setDeleting] = useState(false);
  1187→  const [testingId, setTestingId] = useState<string | null>(null);
  1188→  const [modelConfigTarget, setModelConfigTarget] = useState<LlmProvider | null>(null);
  1189→
  1190→  // Discovery state
  1191→  const [discovering, setDiscovering] = useState(false);
  1192→  const [discovered, setDiscovered] = useState<DiscoveredProvider[]>([]);
  1193→
  1194→  // Fetch providers on mount
  1195→  useEffect(() => {
  1196→    fetchProviders();
  1197→  }, [fetchProviders]);
  1198→
  1199→  // Show error toast
  1200→  useEffect(() => {
  1201→    if (error) {
  1202→      toast.error('Provider error', { description: error });
  1203→    }
  1204→  }, [error]);
  1205→
  1206→  // ---- Handlers ----
  1207→
  1208→  const handleOpenAdd = useCallback(() => {
  1209→    setEditingProvider(null);
  1210→    setFormOpen(true);
  1211→  }, []);
  1212→
  1213→  const handleOpenEdit = useCallback((provider: LlmProvider) => {
  1214→    setEditingProvider(provider);
  1215→    setFormOpen(true);
  1216→  }, []);
  1217→
  1218→  const handleSave = useCallback(
  1219→    async (data: ProviderFormData) => {
  1220→      const typeDef = PROVIDER_TYPES.find((t) => t.value === data.type);
  1221→
  1222→      if (editingProvider) {
  1223→        await updateProvider(editingProvider.id, {
  1224→          name: data.name,
  1225→          type: data.type,
  1226→          baseUrl: data.baseUrl,
  1227→          apiKey: data.apiKey,
  1228→          isLocal: typeDef?.isLocal ?? false,
  1229→          isActive: data.isActive,
  1230→          models: data.models,
  1231→        });
  1232→      } else {
  1233→        await addProvider({
  1234→          name: data.name,
  1235→          type: data.type,
  1236→          baseUrl: data.baseUrl,
  1237→          apiKey: data.apiKey,
  1238→          isLocal: typeDef?.isLocal ?? false,
  1239→          isActive: data.isActive,
  1240→          models: data.models,
  1241→        });
  1242→      }
  1243→    },
  1244→    [editingProvider, addProvider, updateProvider]
  1245→  );
  1246→
  1247→  const handleDelete = useCallback(async () => {
  1248→    if (!deleteTarget) return;
  1249→    setDeleting(true);
  1250→    try {
  1251→      await deleteProvider(deleteTarget.id);
  1252→      toast.success('Provider deleted', {
  1253→        description: `${deleteTarget.name} has been removed.`,
  1254→      });
  1255→      setDeleteTarget(null);
  1256→    } catch (err) {
  1257→      toast.error('Failed to delete provider', {
  1258→        description: err instanceof Error ? err.message : 'Unknown error',
  1259→      });
  1260→    } finally {
  1261→      setDeleting(false);
  1262→    }
  1263→  }, [deleteTarget, deleteProvider]);
  1264→
  1265→  const handleTest = useCallback(
  1266→    async (id: string) => {
  1267→      setTestingId(id);
  1268→      try {
  1269→        const success = await testProvider(id);
  1270→        const provider = providers.find((p) => p.id === id);
  1271→        if (success) {
  1272→          toast.success('Connection successful', {
  1273→            description: `${provider?.name ?? 'Provider'} is reachable and responding.`,
  1274→          });
  1275→        } else {
  1276→          toast.error('Connection failed', {
  1277→            description: `Could not reach ${provider?.name ?? 'provider'}. Check the URL and API key.`,
  1278→          });
  1279→        }
  1280→      } catch {
  1281→        toast.error('Connection test error', {
  1282→          description: 'An unexpected error occurred during the test.',
  1283→        });
  1284→      } finally {
  1285→        setTestingId(null);
  1286→      }
  1287→    },
  1288→    [testProvider, providers]
  1289→  );
  1290→
  1291→  const handleDiscover = useCallback(async () => {
  1292→    setDiscovering(true);
  1293→    setDiscovered([]);
  1294→    try {
  1295→      const res = await fetch('/api/discover');
  1296→      if (!res.ok) {
  1297→        throw new Error('Discovery request failed');
  1298→      }
  1299→      const results: DiscoveredProvider[] = await res.json();
  1300→      setDiscovered(results);
  1301→
  1302→      if (results.length === 0) {
  1303→        toast.info('No local servers found', {
  1304→          description:
  1305→            'Make sure Ollama, LM Studio, vLLM, or another local server is running on its default port.',
  1306→        });
  1307→      } else {
  1308→        toast.success(`Found ${results.length} local ${results.length === 1 ? 'server' : 'servers'}`, {
  1309→          description: 'Select which ones you\'d like to add as providers.',
  1310→        });
  1311→      }
  1312→    } catch (err) {
  1313→      toast.error('Discovery failed', {
  1314→        description: err instanceof Error ? err.message : 'Could not scan for local servers.',
  1315→      });
  1316→    } finally {
  1317→      setDiscovering(false);
  1318→    }
  1319→  }, []);
  1320→
  1321→  const handleAddDiscovered = useCallback(
  1322→    (result: DiscoveredProvider) => {
  1323→      const typeDef = PROVIDER_TYPES.find((t) => t.value === result.type);

  1327→        baseUrl: result.baseUrl,
  1328→        apiKey: '',
  1329→        isLocal: true,
  1330→        isActive: true,
  1331→        models: result.models,
  1332→      }).then(() => {
  1333→        setDiscovered((prev) => prev.filter((r) => r.port !== result.port));
  1334→        toast.success(`${result.name} added`, {
  1335→          description: `Added with ${result.models.length} model(s) on port ${result.port}.`,
  1336→        });
  1337→      }).catch((err) => {
  1338→        toast.error('Failed to add discovered provider', {
  1339→          description: err instanceof Error ? err.message : 'Unknown error',
  1340→        });
  1341→      });
  1342→    },
  1343→    [addProvider]
  1344→  );
  1345→
  1346→  // ---- Render ----
  1347→
  1348→  return (
  1349→    <div className="h-full overflow-y-auto">
  1350→      <div className="max-w-6xl mx-auto p-6 space-y-6">
  1351→        {/* Header */}
  1352→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
  1353→          <div className="flex items-center gap-3">
  1354→            <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-violet-500/10 border border-violet-500/20">
  1355→              <Server className="h-5 w-5 text-violet-400" />
  1356→            </div>
  1357→            <div>
  1358→              <h1 className="text-xl font-bold text-zinc-100 tracking-tight">Model Gateway</h1>
  1359→              <p className="text-sm text-zinc-400 mt-0.5">
  1360→                Configure LLM providers for your agents
  1361→              </p>
  1362→            </div>
  1363→          </div>
  1364→
  1365→          <div className="flex items-center gap-2 shrink-0">
  1366→            <Button
  1367→              variant="outline"
  1368→              onClick={handleDiscover}
  1369→              disabled={discovering}
  1370→              className="border-zinc-700 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100 gap-2"
  1371→            >
  1372→              {discovering ? (
  1373→                <Loader2 className="h-4 w-4 animate-spin" />
  1374→              ) : (
  1375→                <Wifi className="h-4 w-4" />
  1376→              )}
  1377→              Discover Local
  1378→            </Button>

  1380→              onClick={handleOpenAdd}
  1381→              className="bg-violet-600 hover:bg-violet-500 text-white gap-2"

  1383→              <Plus className="h-4 w-4" />

  1386→          </div>
  1387→        </div>
  1388→
  1389→        {/* Discovery results */}
  1390→        <AnimatePresence>

  1392→            <DiscoveryResults

  1395→              onDismiss={() => setDiscovered([])}
  1396→            />

  1400→        {/* Content */}

  1425→      {/* Form Dialog */}
