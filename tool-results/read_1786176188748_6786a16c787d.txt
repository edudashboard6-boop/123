   288→   288→// ---------- Tool Call Card ----------
   289→   289→function ToolCallCard({ toolCall }: { toolCall: ToolCall }) {
   290→   290→  const [copied, setCopied] = useState(false);
   291→   291→  const statusIcon =
   292→   292→    toolCall.status === 'success' ? (
   293→   293→      <CheckCircle className="h-3.5 w-3.5 text-green-400" />
   294→   294→    ) : toolCall.status === 'error' ? (
   295→   295→      <XCircle className="h-3.5 w-3.5 text-red-400" />
   296→   296→    ) : (
   297→   297→      <Clock className="h-3.5 w-3.5 text-muted-foreground/70" />
   298→   298→    );
   299→   299→
   300→   300→  const handleCopy = () => {
   301→   301→    navigator.clipboard.writeText(toolCall.result ?? JSON.stringify(toolCall.arguments, null, 2));
   302→   302→    setCopied(true);
   303→   303→    setTimeout(() => setCopied(false), 2000);
   304→   304→  };
   305→   305→
   306→   306→  return (
   307→   307→    <Collapsible className="my-2">
   308→   308→      <CollapsibleTrigger className="flex w-full items-center gap-2 rounded-lg border border-border bg-card/60 px-3 py-2 text-xs transition-colors hover:bg-muted/60 [&[data-state=open]>svg.chevron]:rotate-90">
   309→   309→        <ChevronRight className="h-3 w-3 text-muted-foreground/70 chevron transition-transform" />
   310→   310→        <Wrench className="h-3 w-3 text-violet-400" />
   311→   311→        <span className="font-medium text-secondary-foreground">{toolCall.name}</span>
   312→   312→        {statusIcon}
   313→   313→        {toolCall.duration != null && (
   314→   314→          <span className="ml-auto text-muted-foreground/50">{toolCall.duration}ms</span>
   315→   315→        )}
   316→   316→      </CollapsibleTrigger>
   317→   317→      <CollapsibleContent className="mt-1 space-y-2 pl-6">
   318→   318→        <div className="rounded-lg border border-border bg-card/40 p-3">
   319→   319→          <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50 mb-1">Arguments</p>
   320→   320→          <pre className="text-xs text-muted-foreground overflow-x-auto whitespace-pre-wrap break-all">
   321→   321→            {JSON.stringify(toolCall.arguments, null, 2)}
   322→   322→          </pre>
   323→   323→        </div>
   324→   324→        {toolCall.result && (
   325→   325→          <div className="rounded-lg border border-border bg-card/40 p-3">
   326→   326→            <div className="flex items-center justify-between mb-1">
   327→   327→              <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50">Result</p>
   328→   328→              <button
   329→   329→                onClick={handleCopy}
   330→   330→                className="text-muted-foreground/50 hover:text-muted-foreground transition-colors"
   331→   331→              >
   332→   332→                {copied ? <Check className="h-3 w-3" /> : <Copy className="h-3 w-3" />}
   333→   333→              </button>
   334→   334→            </div>
   335→   335→            <pre className="text-xs text-muted-foreground overflow-x-auto whitespace-pre-wrap break-all max-h-48">
   336→   336→              {toolCall.result}
   337→   337→            </pre>
   338→   338→          </div>
   339→   339→        )}
   340→   340→      </CollapsibleContent>
   341→   341→    </Collapsible>
   342→   342→  );
   343→   343→}

   676→   676→// ---------- Cost Breakdown ----------
   677→   677→function CostBreakdown({ messages }: { messages: ChatMessage[] }) {
   678→   678→  const agentMap = new Map<string, { tokens: number; cost: number; calls: number }>();
   679→   679→
   680→   680→  messages.forEach((m) => {
   681→   681→    if (m.role === 'system' || m.role === 'user') return;
   682→   682→    const existing = agentMap.get(m.agentName) ?? { tokens: 0, cost: 0, calls: 0 };
   683→   683→    agentMap.set(m.agentName, {
   684→   684→      tokens: existing.tokens + m.tokens,
   685→   685→      cost: existing.cost + m.cost,
   686→   686→      calls: existing.calls + 1,
   687→   687→    });
   688→   688→  });
   689→   689→
   690→   690→  const entries = Array.from(agentMap.entries());
   691→   691→  if (entries.length === 0) return null;
   692→   692→
   693→   693→  const maxTokens = Math.max(...entries.map(([, v]) => v.tokens), 1);
   694→   694→
   695→   695→  return (
   696→   696→    <div className="space-y-2">
   697→   697→      <p className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/50">Cost Breakdown</p>
   698→   698→      <div className="space-y-2.5">
   699→   699→        {entries.map(([name, data]) => {
   700→   700→          const agentDef = DEFAULT_AGENTS.find((a) => a.name === name);
   701→   701→          const pct = (data.tokens / maxTokens) * 100;
   702→   702→          return (
   703→   703→            <div key={name} className="space-y-1">
   704→   704→              <div className="flex items-center justify-between">
   705→   705→                <div className="flex items-center gap-2">
   706→   706→                  <div
   707→   707→                    className={`h-2 w-2 rounded-full ${
   708→   708→                      agentDef ? agentDef.color.replace('text-', 'bg-') : 'bg-accent'
   709→   709→                    }`}
   710→   710→                  />
   711→   711→                  <span className="text-xs text-secondary-foreground">{name}</span>
   712→   712→                </div>
   713→   713→                <div className="flex items-center gap-2 text-[10px] text-muted-foreground/70">
   714→   714→                  <span>{data.tokens.toLocaleString()} tok</span>
   715→   715→                  <span>${data.cost.toFixed(4)}</span>
   716→   716→                </div>
   717→   717→              </div>
   718→   718→              <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   719→   719→                <div
   720→   720→                  className={`h-full rounded-full transition-all ${
   721→   721→                    agentDef ? agentDef.color.replace('text-', 'bg-') : 'bg-accent'
   722→   722→                  }`}
   723→   723→                  style={{ width: `${pct}%` }}
   724→   724→                />
   725→   725→              </div>
   726→   726→            </div>
   727→   727→          );
   728→   728→        })}
   729→   729→      </div>
   730→   730→    </div>
   731→   731→  );
   732→   732→}
