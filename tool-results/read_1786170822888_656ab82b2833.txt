   133→   133→// =============================================================================
   134→   134→// Debounce hook
   135→   135→// =============================================================================
   136→   136→
   137→   137→function useDebouncedCallback<T extends (...args: unknown[]) => unknown>(
   138→   138→  callback: T,
   139→   139→  delay: number,
   140→   140→): T {
   141→   141→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   142→   142→  const callbackRef = useRef(callback);
   143→   143→
   144→   144→  useEffect(() => {
   145→   145→    callbackRef.current = callback;
   146→   146→  }, [callback]);
   147→   147→
   148→   148→  const debounced = useCallback(
   149→   149→    (...args: Parameters<T>) => {
   150→   150→      if (timerRef.current) clearTimeout(timerRef.current);
   151→   151→      timerRef.current = setTimeout(() => callbackRef.current(...args), delay);
   152→   152→    },
   153→   153→    [delay],
   154→   154→  ) as T;
   155→   155→
   156→   156→  useEffect(() => {
   157→   157→    return () => {
   158→   158→      if (timerRef.current) clearTimeout(timerRef.current);
   159→   159→    };
   160→   160→  }, []);
   161→   161→
   162→   162→  return debounced;
   163→   163→}
