   138→// =============================================================================
   139→// Debounce hook
   140→// =============================================================================
   141→
   142→function useDebouncedCallback<T extends (...args: unknown[]) => unknown>(
   143→  callback: T,
   144→  delay: number,
   145→): T {
   146→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   147→  const callbackRef = useRef(callback);
   148→
   149→  useEffect(() => {
   150→    callbackRef.current = callback;
   151→  }, [callback]);
   152→
   153→  const debounced = useCallback(
   154→    (...args: Parameters<T>) => {
   155→      if (timerRef.current) clearTimeout(timerRef.current);
   156→      timerRef.current = setTimeout(() => callbackRef.current(...args), delay);
   157→    },
   158→    [delay],
   159→  ) as T;
   160→
   161→  useEffect(() => {
   162→    return () => {
   163→      if (timerRef.current) clearTimeout(timerRef.current);
   164→    };
   165→  }, []);
   166→
   167→  return debounced;
   168→}
