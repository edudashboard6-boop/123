    75→    75→function formatFileSize(bytes: number): string {
    76→    76→  if (bytes < 1024) return `${bytes} B`;
    77→    77→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    78→    78→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
    79→    79→}
    80→    80→
    81→    81→function relativeTime(dateStr: string): string {
    82→    82→  const now = Date.now();
    83→    83→  const then = new Date(dateStr).getTime();
    84→    84→  const diff = now - then;
    85→    85→  const mins = Math.floor(diff / 60000);
    86→    86→  if (mins < 1) return 'just now';
    87→    87→  if (mins < 60) return `${mins}m ago`;
    88→    88→  const hrs = Math.floor(mins / 60);
    89→    89→  if (hrs < 24) return `${hrs}h ago`;
    90→    90→  const days = Math.floor(hrs / 24);
    91→    91→  if (days < 30) return `${days}d ago`;
    92→    92→  return new Date(dateStr).toLocaleDateString();
    93→    93→}

    95→    95→function fileIconForType(fileType: string) {
    96→    96→  if (fileType === 'json') return FileJson;
    97→    97→  if (['py', 'js', 'ts', 'csv'].includes(fileType)) return FileCode;
    98→    98→  return FileText;
    99→    99→}

   101→   101→function categoryColor(category: string): string {
   102→   102→  switch (category) {
   103→   103→    case 'text': return 'bg-zinc-700/50 text-zinc-300 border-zinc-600/50';
   104→   104→    case 'code': return 'bg-emerald-900/30 text-emerald-400 border-emerald-700/40';
   105→   105→    case 'data': return 'bg-amber-900/30 text-amber-400 border-amber-700/40';
   106→   106→    default: return 'bg-violet-900/30 text-violet-400 border-violet-700/40';
   107→   107→  }
   108→   108→}

   114→   114→}
