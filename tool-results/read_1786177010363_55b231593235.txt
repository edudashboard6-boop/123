   100→
   101→// -----------------------------------------------------------------------------
   102→// Component
   103→// -----------------------------------------------------------------------------
   104→
   105→export function SettingsPanel() {
   106→  const { setTheme } = useTheme();
   107→  const [settings, setSettings] = useState<AppSettings>(DEFAULT_SETTINGS);
   108→  const [loading, setLoading] = useState(true);
   109→  const [saving, setSaving] = useState(false);
   110→  const [saveFlash, setSaveFlash] = useState<string | null>(null);
   111→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   112→
   113→  // ---------------------------------------------------------------------------
   114→  // Fetch settings
   115→  // ---------------------------------------------------------------------------
   116→
   117→  const fetchSettings = useCallback(async () => {
   118→    setLoading(true);
   119→    try {
   120→      const res = await fetch('/api/settings');
   121→      if (res.ok) {
   122→        const data = await res.json();
   123→        setSettings({ ...DEFAULT_SETTINGS, ...data });
   124→      }
   125→    } catch {
   126→      // use defaults
   127→    } finally {
   128→      setLoading(false);
   129→    }
   130→  }, []);
   131→
   132→  useEffect(() => {
   133→    fetchSettings();
   134→  }, [fetchSettings]);
   135→
   136→  // ---------------------------------------------------------------------------
   137→  // Save settings (debounced)
   138→  // ---------------------------------------------------------------------------
   139→
   140→  const saveSettings = useCallback(async (newSettings: AppSettings) => {
   141→    setSaving(true);
   142→    try {
   143→      const res = await fetch('/api/settings', {
   144→        method: 'PATCH',
   145→        headers: { 'Content-Type': 'application/json' },
   146→        body: JSON.stringify(newSettings),
   147→      });
   148→      if (!res.ok) throw new Error('Save failed');
   149→    } catch {
   150→      toast.error('Failed to save settings');
   151→    } finally {
   152→      setSaving(false);
   153→    }
   154→  }, []);
   155→
   156→  const updateSetting = useCallback(
   157→    <K extends keyof AppSettings>(key: K, value: AppSettings[K]) => {
   158→      setSettings((prev) => {
   159→        const updated = { ...prev, [key]: value };
   160→        // Debounced save
   161→        if (debounceRef.current) clearTimeout(debounceRef.current);
   162→        debounceRef.current = setTimeout(() => {
   163→          saveSettings(updated);
   164→        }, 500);
   165→        return updated;
   166→      });
   167→    },
   168→    [saveSettings],
   169→  );
   170→
   171→  const updateAgentBudget = useCallback(
   172→    (agentId: string, value: number) => {
   173→      setSettings((prev) => {
   174→        const updated = {
   175→          ...prev,
   176→          agentBudgetLimits: { ...prev.agentBudgetLimits, [agentId]: value },
   177→        };
   178→        if (debounceRef.current) clearTimeout(debounceRef.current);
   179→        debounceRef.current = setTimeout(() => {
   180→          saveSettings(updated);
   181→        }, 500);
   182→        return updated;
   183→      });
   184→    },
   185→    [saveSettings],
   186→  );
   187→
   188→  const flashSave = useCallback((key: string) => {
   189→    setSaveFlash(key);
   190→    setTimeout(() => setSaveFlash(null), 1500);
   191→  }, []);
   192→
   193→  const handleResetCosts = useCallback(() => {
   194→    toast.success('Cost counters have been reset');
   195→  }, []);
   196→
   197→  // ---------------------------------------------------------------------------
   198→  // Render helpers
   199→  // ---------------------------------------------------------------------------
   200→
   201→  const renderSaveIndicator = (key: string) => {
   202→    if (saveFlash === key) {
   203→      return (
   204→        <span className="flex items-center gap-1 text-[10px] text-emerald-400">
   205→          <Check className="h-3 w-3" /> Saved
   206→        </span>
   207→      );
   208→    }
   209→    if (saving) {
   210→      return (
   211→        <span className="text-[10px] text-muted-foreground/70">Saving...</span>
   212→      );
   213→    }
   214→    return null;
   215→  };
   216→
   217→  // ---------------------------------------------------------------------------
   218→  // Render
   219→  // ---------------------------------------------------------------------------
   220→

   223→      <div className="flex h-full flex-col min-h-0">
   224→        {/* Header */}
   225→        <div className="flex flex-col gap-1 border-b border-border/60 px-6 py-4">
   226→          <div className="flex items-center justify-between">
   227→            <div className="flex items-center gap-3">
   228→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-violet-600/15 border border-violet-500/20">
   229→                <Settings className="h-4.5 w-4.5 text-violet-400" />
   230→              </div>
   231→              <div>
   232→                <h1 className="text-base font-semibold text-foreground">Settings</h1>
   233→                <p className="text-xs text-muted-foreground/70">Configure system preferences and orchestration parameters</p>
   234→              </div>
   235→            </div>
   236→          </div>
   237→        </div>
   238→
   239→        <ScrollArea className="flex-1 min-h-0">
   240→          <div className="mx-auto w-full max-w-3xl space-y-6 p-6">
   241→
   242→            {/* General Settings */}
   243→            <Card className="bg-card/80 border-border/60">
   244→              <CardHeader className="pb-3">
   245→                <CardTitle className="text-sm font-semibold text-secondary-foreground">General</CardTitle>
   246→                <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   247→                  Appearance, language, and default behavior
   248→                </CardDescription>
   249→              </CardHeader>
   250→              <CardContent className="space-y-5">
   251→                {/* Theme */}
   252→                <div className="flex items-center justify-between">
   253→                  <div className="flex items-center gap-3">
   254→                    <div className="flex h-8 w-8 items-center justify-center rounded-lg">
   255→                      {settings.theme === 'dark' ? (
   256→                        <Moon className="h-4 w-4 text-violet-400" />
   257→                      ) : (
   258→                        <Sun className="h-4 w-4 text-amber-400" />
   259→                      )}
   260→                    </div>
   261→                    <div>
   262→                      <Label className="text-xs text-secondary-foreground">Theme</Label>
   263→                      <p className="text-[11px] text-muted-foreground/50">Switch between light and dark mode</p>
   264→                    </div>
   265→                  </div>
   266→                  <Select
   267→                    value={settings.theme}
   268→                    onValueChange={(v) => {
   269→                      const theme = v as 'dark' | 'light';
   270→                      setTheme(theme);
   271→                      updateSetting('theme', theme);
   272→                    }}
   273→                  >
   274→                    <SelectTrigger className="w-32 h-8">
   275→                      <SelectValue />
   276→                    </SelectTrigger>
   277→                    <SelectContent className="bg-popover border-border">
   278→                      <SelectItem value="dark" className="text-secondary-foreground">
   279→                        <div className="flex items-center gap-2"><Moon className="h-3 w-3" /> Dark</div>
   280→                      </SelectItem>

   322→                      <SelectValue />
   323→                    </SelectTrigger>
   324→                    <SelectContent className="bg-popover border-border">
   325→                      <SelectItem value="fast" className="text-secondary-foreground">Fast</SelectItem>

   329→                  </Select>
   330→                </div>
   331→
   332→                <Separator  />

   334→                {/* Auto-save */}
   335→                <div className="flex items-center justify-between">
   336→                  <div>
   337→                    <Label className="text-xs text-secondary-foreground">Auto-save</Label>
   338→                    <p className="text-[11px] text-muted-foreground/50">Automatically save chat sessions</p>
   339→                  </div>
   340→                  <Switch
   341→                    checked={settings.autoSave}
   342→                    onCheckedChange={(v) => { updateSetting('autoSave', v); flashSave('autoSave'); }}
   343→                    className="data-[state=checked]:bg-violet-600"
   344→                  />
   345→                </div>
   346→              </CardContent>
   347→            </Card>
   348→
   349→            {/* Cost Control */}
   350→            <Card className="bg-card/80 border-border/60">
   351→              <CardHeader className="pb-3">
   352→                <div className="flex items-center justify-between">
   353→                  <div>
   354→                    <CardTitle className="text-sm font-semibold text-secondary-foreground">Cost Control</CardTitle>
   355→                    <CardDescription className="text-xs text-muted-foreground/70 mt-0.5">
   356→                      Budget limits and spending alerts
   357→                    </CardDescription>
   358→                  </div>
   359→                  <Button
   360→                    variant="outline"
   361→                    size="sm"
   362→                    className="h-7 text-[11px] text-muted-foreground hover:text-secondary-foreground border-input hover:"
   363→                    onClick={handleResetCosts}
   364→                  >
   365→                    <RotateCcw className="h-3 w-3" />
   366→                    Reset Counters
   367→                  </Button>
   368→                </div>
   369→              </CardHeader>
   370→              <CardContent className="space-y-5">
   371→                {/* Per-session budget */}
   372→                <div className="space-y-2">
   373→                  <div className="flex items-center justify-between">
   374→                    <Label className="text-xs text-secondary-foreground">Per-Session Budget Limit</Label>

   379→                    <Input
   380→                      type="number"
   381→                      min={0}
   382→                      step={1}
   383→                      value={settings.sessionBudgetLimit}
   384→                      onChange={(e) => updateSetting('sessionBudgetLimit', parseFloat(e.target.value) || 0)}
   385→                      onBlur={() => flashSave('sessionBudget')}
   386→                      className="h-8 w-24"
   387→                    />

   389→                  </div>
   390→                </div>
   391→
   392→                <Separator  />
   393→
   394→                {/* Per-agent budget limits */}
   395→                <div className="space-y-3">
   396→                  <Label className="text-xs text-secondary-foreground">Per-Agent Budget Limits</Label>
   397→                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   398→                    {Object.entries(AGENT_LABELS).map(([agentId, { name, color }]) => (
   399→                      <div key={agentId} className="flex items-center justify-between rounded-lg">
   400→                        <span className={`text-xs font-medium ${color}`}>{name}</span>

   404→                            type="number"
   405→                            min={0}
   406→                            step={0.5}
   407→                            value={settings.agentBudgetLimits[agentId] ?? 0}
   408→                            onChange={(e) => updateAgentBudget(agentId, parseFloat(e.target.value) || 0)}
   409→                            onBlur={() => flashSave(`agent-${agentId}`)}
   410→                            className="h-7 w-16"
   411→                          />
   412→                        </div>
   413→                      </div>

   451→                      Sandbox, tool approvals, and audit settings

   453→                  </div>
   454→                </div>
   455→              </CardHeader>
   456→              <CardContent className="space-y-5">

   466→                    className="data-[state=checked]:bg-violet-600"

   472→                {/* Auto-approve Tools */}

   474→                  <div>
   475→                    <Label className="text-xs text-secondary-foreground">Auto-approve Tools</Label>
