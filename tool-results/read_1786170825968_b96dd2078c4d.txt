   133→// =============================================================================
   134→// Debounce hook
   135→// =============================================================================
   136→
   137→function useDebouncedCallback<T extends (...args: unknown[]) => unknown>(
   138→  callback: T,
   139→  delay: number,
   140→): T {
   141→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   142→  const callbackRef = useRef(callback);
   143→
   144→  useEffect(() => {
   145→    callbackRef.current = callback;
   146→  }, [callback]);
   147→
   148→  const debounced = useCallback(
   149→    (...args: Parameters<T>) => {
   150→      if (timerRef.current) clearTimeout(timerRef.current);
   151→      timerRef.current = setTimeout(() => callbackRef.current(...args), delay);
   152→    },
   153→    [delay],
   154→  ) as T;
   155→
   156→  useEffect(() => {
   157→    return () => {
   158→      if (timerRef.current) clearTimeout(timerRef.current);
   159→    };
   160→  }, []);
   161→
   162→  return debounced;
   163→}
